// CE10295 refactor: the strict CISD run tracker (V2 FIX-16, unchanged logic)
// keeps its run state inside a function and returns the two events plus the
// two run lengths the debug table shows.
f_cisd() =>
    var float _bullRunOpen = na
    var float _bearRunOpen = na
    var int   _bullRunLen  = 0
    var int   _bearRunLen  = 0
    var bool  _cisdDnDone  = false
    var bool  _cisdUpDone  = false
    if confirmed
        if cisdNeutral
            _bullRunOpen := na
            _bearRunOpen := na
            _bullRunLen  := 0
            _bearRunLen  := 0
        if cisdBull and not cisdBull[1]
            _bullRunOpen := open
            _bullRunLen  := 1
            _cisdDnDone  := false
        else if cisdBull
            _bullRunLen += 1
        if cisdBear and not cisdBear[1]
            _bearRunOpen := open
            _bearRunLen  := 1
            _cisdUpDone  := false
        else if cisdBear
            _bearRunLen += 1
    _cisdDn = confirmed and not na(_bullRunOpen) and not _cisdDnDone and cisdBear and close < _bullRunOpen and _bullRunLen >= 2 and body >= BRK_BODY_ATR * atr and close <= low + STRONG_CLOSE_PCT * rng
    _cisdUp = confirmed and not na(_bearRunOpen) and not _cisdUpDone and cisdBull and close > _bearRunOpen and _bearRunLen >= 2 and body >= BRK_BODY_ATR * atr and close >= high - STRONG_CLOSE_PCT * rng
    if _cisdDn
        _cisdDnDone := true
    if _cisdUp
        _cisdUpDone := true
    if showCisd and _cisdDn
        f_chip(false, "CISD", color.new(COL_BEAR, 55), color.white, mkSize)
    if showCisd and _cisdUp
        f_chip(true, "CISD", color.new(COL_BULL, 55), color.white, mkSize)
    [_cisdDn, _cisdUp, _bullRunLen, _bearRunLen]

    var int   dir    = 0

    var float protHi = na
    var float legLo  = na
    var float legHi  = na

        else if dnBrk

            dir    := -1
            protHi := legHi

            legHi := high
    [dir, bosU, bosD, chU, chD, protLo, protHi]

// turtle-soup age note (audited, V2 logic KEPT): ta.lowestbars/highestbars
// return a NON-POSITIVE offset (0 = this bar, -k = k bars back). Taking the
// value on the [1] series and negating: -offset[1] + 1 = positive age of the
// extreme in bars as seen from the CURRENT bar (≥ 1). No sign bug. v3 adds
// the MAX age gate: fading a months-old level is not a turtle soup.
f_tbsAgeOk(int age) => age >= tbsAgeMin and (tbsAgeMax == 0 or age <= tbsAgeMax)

f_crtTbsHtf() =>
    _atr  = ta.atr(atrLen)
    _rng  = high - low
    _body = math.abs(close - open)
    _bull = close > open
    _bear = close < open
    _c1big = _rng[2] >= crtRangeAtr * _atr
    _crtB  = _c1big and low[1] < low[2] and close[1] > low[2] and close[1] < high[2] and low[2] - low[1] >= minWickAtr * _atr and _bull and close > high[1] and _body >= crtDispAtr * _atr
    _crtS  = _c1big and high[1] > high[2] and close[1] < high[2] and close[1] > low[2] and high[1] - high[2] >= minWickAtr * _atr and _bear and close < low[1] and _body >= crtDispAtr * _atr
    _tl    = ta.lowest(low, tbsLen)[1]
    _th    = ta.highest(high, tbsLen)[1]
    _loAge = -ta.lowestbars(low, tbsLen)[1] + 1
    _hiAge = -ta.highestbars(high, tbsLen)[1] + 1
    _tbsB  = low < _tl and close > _tl and _bull and f_tbsAgeOk(_loAge) and close >= low + 0.5 * _rng
    _tbsS  = high > _th and close < _th and _bear and f_tbsAgeOk(_hiAge) and close <= high - 0.5 * _rng
    [_crtB[1], _crtS[1], _tbsB[1], _tbsS[1], time[1]]

f_htfPoi() =>

    var float bBot = na

    var float sBot = na
    var int   sBar = na

        bBot := high[2]

        sBot := high
        sTop := low[2]

        sTop := na
        sBot := na
    [bTop[1], bBot[1], sTop[1], sBot[1], bIn[1], sIn[1]]

f_insideBar() =>

    var float _mHigh   = na
    var float _mLow    = na
    var float _ibHigh  = na
    var float _ibLow   = na
    var int   _ibStart = na
    _sig3Long  = false
    _sig3Short = false
    _mMid = (_mHigh + _mLow) / 2
    if confirmed
        if _ibStage == 1
            if high < _mHigh and low > _mLow and low >= _mMid
                _ibHigh := high
                _ibLow  := low
                _ibStage := 2
            else
                _ibStage := 0
        else if _ibStage == 2
            if close > _ibHigh
                if bullBar and body >= BRK_BODY_ATR * atr and close >= high - STRONG_CLOSE_PCT * rng and structDir >= 0 and htfB >= 0
                    _sig3Long := showS3
                _ibStage := 0
            else if close < _ibLow or bar_index - _ibStart > IB_WAIT_BARS
                _ibStage := 0
        else if _ibStage == -1
            if high < _mHigh and low > _mLow and high <= _mMid
                _ibHigh := high
                _ibLow  := low
                _ibStage := -2
            else
                _ibStage := 0
        else if _ibStage == -2
            if close < _ibLow
                if bearBar and body >= BRK_BODY_ATR * atr and close <= low + STRONG_CLOSE_PCT * rng and structDir <= 0 and htfB <= 0
                    _sig3Short := showS3
                _ibStage := 0
            else if close > _ibHigh or bar_index - _ibStart > IB_WAIT_BARS
                _ibStage := 0
        if showS3 and _ibStage == 0 and brokeUp and body >= ibBodyAtr * atr
            _ibStage := 1
            _mHigh := high
            _mLow  := low
            _ibStart := bar_index
        else if showS3 and _ibStage == 0 and brokeDn and body >= ibBodyAtr * atr
            _ibStage := -1
            _mHigh := high
            _mLow  := low
            _ibStart := bar_index
    [_sig3Long, _sig3Short]

// CE10295 refactor: f_ltScore is lifted to global scope (Pine has no nested
// function definitions) and now takes the two machine states as parameters.
// confidence score (§12) — clamped 0–100, thresholds configurable (§13)
f_ltScore(bool isLong, bool retOk, int lState, int sState) =>
    int s = 10                                                                              // key-level interaction (anchored by construction)
    s += (isLong ? sslInPlay : bslInPlay) ? 20 : 0                                          // liquidity sweep in play
    bool strongRej = isLong ? (bullPin or (rng > 0 and dnWick >= 2 * body and dnWick >= minWickAtr * atr)) : (bearPin or (rng > 0 and upWick >= 2 * body and upWick >= minWickAtr * atr))
    s += strongRej ? 15 : 0                                                                 // strong rejection candle
    s += (isLong ? (recentMssUp or recentChUp) : (recentMssDn or recentChDn)) ? 20 : 0      // MSS / CHoCH
    s += (isLong ? (inDemand or nearDemand) : (inSupply or nearSupply)) ? 10 : 0            // S/D confirmation
    s += (isLong ? not na(cFvgBTop) : not na(cFvgSTop)) ? 10 : 0                            // live FVG on signal side
    s += (isLong ? obNearL : obNearS) ? 10 : 0                                              // order-block interaction
    s += (isLong ? recentDispUp : recentDispDn) ? 10 : 0                                    // fresh displacement
    s += retOk ? 10 : 0                                                                     // retest / failed-break done
    s += (isLong ? htfB > 0 : htfB < 0) ? 15 : 0                                            // MTF agreement
    s -= regime == "CHOP" ? 10 : 0                                                          // weak / unclear structure
    s -= (isLong ? extDir == -1 : extDir == 1) ? 10 : 0                                     // counter-trend
    s -= ltLowVol ? 5 : 0                                                                   // low volatility
    s -= not (isLong ? (sslInPlay or recentEngLo) : (bslInPlay or recentEngHi)) ? 10 : 0    // insufficient liquidity
    s -= lState >= 1 and sState >= 1 ? 15 : 0                                           // ambiguous — both directions armed
    math.max(0, math.min(100, s))

        if not na(suLSweepExt) and close < suLSweepExt
            suL := 0
            suLReason := "sweep undone"
        else if bearBar and body >= dispFactor * atr and suL <= 3
            suL := 0
            suLReason := "opposing displacement"
    if suL >= 2 and zoneInvalidatedUp and na(suLFvgBar)
        suL := 0
        suLReason := "POI violated"
    if sbKillAtEnd and sbKillL and suL >= 1 and suL <= 3
        suL := 0
        suLReason := "SB window closed"
    // stage expiry
    if suL == 1 and bar_index - suLBar > sbLiqWL
        suL := 0
        suLReason := "sweep expired (no POI tap)"
    if suL == 2 and bar_index - suLBar > sbPoiWL
        suL := 0
        suLReason := "POI tap expired (no shift)"
    if suL == 3 and bar_index - suLBar > sbRetWL
        suL := 0
        suLReason := "retest window expired"
    // transitions
    // ◆ SB arming path: `or sbArmOkL` lets the SB gate arm a setup that the
    // generic minArmQual would reject, and the chop block may be waived for it.
    if suL == 0 and sslInPlay and sbArmBarL and (lastSslQual >= minArmQual or sbArmOkL)
        bool chopBlock = useRegime and regime == "CHOP" and lastSslQual < 4 and not (sbArmOkL and sbChopBypass)
        if not chopBlock
            suL := 1
            suLBar := bar_index
            suLSweepLvl := lastSslLvl
            suLSweepExt := lastSslExt
            suLAnchorT := na
            suLAnchorB := na
            suLFvgBar  := na
            sbTagL     := sbArmOkL ? sbCtxWin : 0
            sbShiftGL  := 0
            if sbArmOkL
                sbArmedSslBar := lastSslBar     // SB-9: this sweep is spent
    if suL == 1
        bool tapZone = inDemand and retestLong
        bool tapOte  = inOTEbuy
        bool tapHtf  = htfLive and not na(hPoiBTop) and low <= hPoiBTop and close > hPoiBBot
        if tapZone or tapOte or tapHtf
            suL := 2
            suLBar := bar_index
            if tapZone
                suLAnchorT := nbTop
                suLAnchorB := nbBot
    // ◆ SB: MSS preferred. With sbReqMss on, a tagged SB setup only shifts on a
    // true MSS; otherwise an external CHoCH or a CISD still qualifies (its lower
    // grade is recorded in sbShiftGL and costs a point in the SB score).
    if suL == 2 and (sbTagL != 0 and sbReqMss ? mssUp : (mssUp or chochUp or eChochUp or cisdUp))
        suL := 3
        suLBar := bar_index
        sbShiftGL := mssUp ? 2 : 1
        // a TAGGED SB setup may only lock a CLEAN displacement FVG
        if not na(cFvgBTop) and cFvgBBar >= suLBar - TRIG_LOOK and (sbTagL == 0 or sbFvgOkL)
            suLAnchorT := cFvgBTop      // lock the shift-leg FVG as the anchor
            suLAnchorB := cFvgBBot
            suLFvgBar  := cFvgBBar
    if suL == 3
        if na(suLFvgBar) and not na(cFvgBTop) and cFvgBBar > suLBar and (sbTagL == 0 or sbFvgOkL)
            suLAnchorT := cFvgBTop      // first FVG formed after the shift
            suLAnchorB := cFvgBBot
            suLFvgBar  := cFvgBBar
        if not na(suLAnchorT)
            float ceL       = (suLAnchorT + suLAnchorB) / 2      // consequent encroachment
            bool  ripeL     = na(suLFvgBar) or bar_index > suLFvgBar
            // SB needs a LOCKED FVG anchor and (by default) an in-window entry
            bool  sbSetL    = sbTagL != 0 and not na(suLFvgBar) and (sbWinNow == sbTagL or not sbKillAtEnd)
            // ◆ SB-11: a CE fill is a TOUCH plus CONFIRMATION. Touching the CE
            // with a wick and closing back below it is price REJECTING the
            // level, not accepting it — that fill is refused unless the bar
            // itself is a genuine rejection (pin / engulf / IFC / CISD).
            bool  sbRejL    = bullPin or bullEngulf or ifcBull or cisdUp
            bool  sbCeTchL  = low <= suLAnchorT and low <= ceL + sbCeTol * atr
            bool  sbCeTapL  = sbCeTchL and (close >= ceL or sbRejL)
            bool  sbPrxTapL = low <= suLAnchorT and sbRejL
            // ◆ SB-12: premium / discount off the LOCKED window range
            bool  sbLocOkL  = (not sbReqHtf or htfB > 0) and (not sbReqPD or (sbPdOK and sbPdPos < 0.5))
            bool  sbEnterL  = sbSetL and ripeL and sbLocOkL and (sbCeTapL or (sbEntryMode != "CE (50%) only" and sbPrxTapL))
            bool  stdEnterL = sbTagL == 0 and ripeL and low <= suLAnchorT and zoneBullConf
            if close < suLAnchorB
                suL := 0
                suLReason := sbTagL != 0 ? "SB FVG violated" : "anchor FVG violated"
            else if (sbEnterL or stdEnterL) and close > suLAnchorB
                sigMCLong := true
                sbSigLong := sbEnterL
                sbCeFillL := sbEnterL and sbCeTapL
                sigSLLong := math.min(math.min(low, suLAnchorB), nz(suLSweepExt, low)) - SL_PAD_ATR * atr
                suL := 4
                suLBar := bar_index

// SHORT machine
if confirmed
    if suS == 0
        sbTagS    := 0
        sbShiftGS := 0
    if suS >= 1
        if not na(suSSweepExt) and close > suSSweepExt
            suS := 0
            suSReason := "sweep undone"
        else if bullBar and body >= dispFactor * atr and suS <= 3
            suS := 0
            suSReason := "opposing displacement"
    if suS >= 2 and zoneInvalidatedDn and na(suSFvgBar)
        suS := 0
        suSReason := "POI violated"
    if sbKillAtEnd and sbKillS and suS >= 1 and suS <= 3
        suS := 0
        suSReason := "SB window closed"
    if suS == 1 and bar_index - suSBar > sbLiqWS
        suS := 0
        suSReason := "sweep expired (no POI tap)"
    if suS == 2 and bar_index - suSBar > sbPoiWS
        suS := 0
        suSReason := "POI tap expired (no shift)"
    if suS == 3 and bar_index - suSBar > sbRetWS
        suS := 0
        suSReason := "retest window expired"
    if suS == 0 and bslInPlay and sbArmBarS and (lastBslQual >= minArmQual or sbArmOkS)
        bool chopBlock = useRegime and regime == "CHOP" and lastBslQual < 4 and not (sbArmOkS and sbChopBypass)
        if not chopBlock
            suS := 1
            suSBar := bar_index
            suSSweepLvl := lastBslLvl
            suSSweepExt := lastBslExt
            suSAnchorT := na
            suSAnchorB := na
            suSFvgBar  := na
            sbTagS     := sbArmOkS ? sbCtxWin : 0
            sbShiftGS  := 0
            if sbArmOkS
                sbArmedBslBar := lastBslBar     // SB-9: this sweep is spent
    if suS == 1
        bool tapZone = inSupply and retestShort
        bool tapOte  = inOTEsell
        bool tapHtf  = htfLive and not na(hPoiSTop) and high >= hPoiSBot and close < hPoiSTop
        if tapZone or tapOte or tapHtf
            suS := 2
            suSBar := bar_index
            if tapZone
                suSAnchorT := nsTop
                suSAnchorB := nsBot
    if suS == 2 and (sbTagS != 0 and sbReqMss ? mssDn : (mssDn or chochDn or eChochDn or cisdDn))
        suS := 3
        suSBar := bar_index
        sbShiftGS := mssDn ? 2 : 1
        if not na(cFvgSTop) and cFvgSBar >= suSBar - TRIG_LOOK and (sbTagS == 0 or sbFvgOkS)
            suSAnchorT := cFvgSTop
            suSAnchorB := cFvgSBot
            suSFvgBar  := cFvgSBar
    if suS == 3
        if na(suSFvgBar) and not na(cFvgSTop) and cFvgSBar > suSBar and (sbTagS == 0 or sbFvgOkS)
            suSAnchorT := cFvgSTop
            suSAnchorB := cFvgSBot
            suSFvgBar  := cFvgSBar
        if not na(suSAnchorB)
            float ceS       = (suSAnchorT + suSAnchorB) / 2
            bool  ripeS     = na(suSFvgBar) or bar_index > suSFvgBar
            bool  sbSetS    = sbTagS != 0 and not na(suSFvgBar) and (sbWinNow == sbTagS or not sbKillAtEnd)
            // ◆ SB-11: touch + close through the CE, or a genuine rejection
            bool  sbRejS    = bearPin or bearEngulf or ifcBear or cisdDn
            bool  sbCeTchS  = high >= suSAnchorB and high >= ceS - sbCeTol * atr
            bool  sbCeTapS  = sbCeTchS and (close <= ceS or sbRejS)
            bool  sbPrxTapS = high >= suSAnchorB and sbRejS
            // ◆ SB-12: premium / discount off the LOCKED window range
            bool  sbLocOkS  = (not sbReqHtf or htfB < 0) and (not sbReqPD or (sbPdOK and sbPdPos >= 0.5))
            bool  sbEnterS  = sbSetS and ripeS and sbLocOkS and (sbCeTapS or (sbEntryMode != "CE (50%) only" and sbPrxTapS))
            bool  stdEnterS = sbTagS == 0 and ripeS and high >= suSAnchorB and zoneBearConf
            if close > suSAnchorT
                suS := 0
                suSReason := sbTagS != 0 ? "SB FVG violated" : "anchor FVG violated"
            else if (sbEnterS or stdEnterS) and close < suSAnchorT
                sigMCShort := true
                sbSigShort := sbEnterS
                sbCeFillS  := sbEnterS and sbCeTapS
                sigSLShort := math.max(math.max(high, suSAnchorT), nz(suSSweepExt, high)) + SL_PAD_ATR * atr
                suS := 4
                suSBar := bar_index

// SIGNAL FINALIZE — context gates, tier filter, cooldowns, conflict rule
// Precedence on conflict (spec §45): the machine narrative wins; if both
// directions fire on one bar → NO TRADE (both suppressed).
// ═════════════════════════════════════════════════════════════════════════
var int lastLongSigBar  = na
var int lastShortSigBar = na
longCdOK  = masterCd == 0 or na(lastLongSigBar)  or bar_index - lastLongSigBar  > masterCd
shortCdOK = masterCd == 0 or na(lastShortSigBar) or bar_index - lastShortSigBar > masterCd

chopOKL = not useRegime or regime != "CHOP" or lastSslQual >= 4
chopOKS = not useRegime or regime != "CHOP" or lastBslQual >= 4
// ◆ SB-only mode: nothing fires outside an SB context window
sbOnlyOK = not (sbEnable and sbOnly and timeframe.isintraday) or sbCtx
longOK  = (biasFilter == "Off" or filtBias >= 0) and kzOK and newsOK and longCdOK and chopOKL and sbOnlyOK and (minScore == 0 or scoreLong >= minScore)
shortOK = (biasFilter == "Off" or filtBias <= 0) and kzOK and newsOK and shortCdOK and chopOKS and sbOnlyOK and (minScore == 0 or scoreShort >= minScore)

// machine entries: tier is A by construction, A+ when the extras align
mcTierL = math.max(tierLong, 3)
mcTierS = math.max(tierShort, 3)
// ◆ SB-7: a pure SB entry must clear a HIGHER tier floor than the generic
// machine entry (A by default, A+ by input) on top of the display filter.
sbTierMin = sbMinTier == "A+ only" ? 4 : 3
sigMCLong  := sigMCLong  and longOK  and mcTierL >= (sbSigLong  ? math.max(minRank, sbTierMin) : minRank)
sigMCShort := sigMCShort and shortOK and mcTierS >= (sbSigShort ? math.max(minRank, sbTierMin) : minRank)
if not sigMCLong
    sbSigLong := false
if not sigMCShort
    sbSigShort := false

// engine trigger candidates: graded through the same layer model
// ◆ SB precedence: while a clean SB setup is live (tagged and past its POI)
// the ordinary engine candidates are down-weighted one tier, or muted. The
// machine path itself is never muted — it is what carries the SB narrative.
sbLiveL = sbEnable and sbTagL != 0 and suL >= 2
sbLiveS = sbEnable and sbTagS != 0 and suS >= 2
sbLive  = sbLiveL or sbLiveS
engRank = sbLive and sbSuppress == "Down-weight" ? math.min(4, minRank + 1) : minRank
engMute = sbLive and sbSuppress == "Suppress"
engTierOKL = tierLong  >= engRank and not engMute
engTierOKS = tierShort >= engRank and not engMute
sigSDLong   := sigSDLong   and longOK and engTierOKL
sig3Long    := sig3Long    and longOK and engTierOKL
sig4Long    := sig4Long    and longOK and engTierOKL
sigTLLong   := sigTLLong   and longOK and engTierOKL
sigCRTLong  := sigCRTLong  and longOK and engTierOKL
sigTBSLong  := sigTBSLong  and longOK and engTierOKL
sigHCrtLong := sigHCrtLong and longOK and engTierOKL
sigHTbsLong := sigHTbsLong and longOK and engTierOKL
sigFEMLong  := sigFEMLong  and longOK and engTierOKL
sigSDShort   := sigSDShort   and shortOK and engTierOKS
sig3Short    := sig3Short    and shortOK and engTierOKS
sig4Short    := sig4Short    and shortOK and engTierOKS
sigTLShort   := sigTLShort   and shortOK and engTierOKS
sigCRTShort  := sigCRTShort  and shortOK and engTierOKS
sigTBSShort  := sigTBSShort  and shortOK and engTierOKS
sigHCrtShort := sigHCrtShort and shortOK and engTierOKS
sigHTbsShort := sigHTbsShort and shortOK and engTierOKS
sigFEMShort  := sigFEMShort  and shortOK and engTierOKS

anyLong  = sigMCLong  or sigSDLong  or sig3Long  or sig4Long  or sigTLLong  or sigCRTLong  or sigTBSLong  or sigHCrtLong  or sigHTbsLong  or sigFEMLong
anyShort = sigMCShort or sigSDShort or sig3Short or sig4Short or sigTLShort or sigCRTShort or sigTBSShort or sigHCrtShort or sigHTbsShort or sigFEMShort

// ◆ SB-13 conflict rule — a HIERARCHY, not a mutual kill. v3 muted BOTH sides
// whenever the two directions fired on one bar, so a full SB narrative was
// thrown away by any stray engine candidate on the other side (the "silent
// day" problem). Rank of a bar's signal:
//     3 = SB machine entry · 2 = ordinary machine entry · 1 = pure engine
// The higher rank takes the bar and the loser is muted. Only an EQUAL-rank
// clash still mutes both — engine vs engine, machine vs machine, SB vs SB —
// because that is a genuinely split narrative with no tie-break left.
if anyLong and anyShort
    int rkL = sigMCLong  ? (sbSigLong  ? 3 : 2) : 1
    int rkS = sigMCShort ? (sbSigShort ? 3 : 2) : 1
    if rkL > rkS
        anyShort   := false
        sigMCShort := false
        sbSigShort := false
        if suS == 4 and suSBar == bar_index     // roll back the losing entry
            suS := 0
            suSReason := "conflict — higher-rank long won the bar"
    else if rkS > rkL
        anyLong   := false
        sigMCLong := false
        sbSigLong := false
        if suL == 4 and suLBar == bar_index
            suL := 0
            suLReason := "conflict — higher-rank short won the bar"
    else
        anyLong  := false
        anyShort := false

// ◬ Liquidity Trap veto (opt-in, default OFF — backward compatible): while the
// trap engine is WAITING for confirmation in one direction, the opposing
// breakout side is suspect — suppress those engine signals until it resolves.
if ltEnable and ltVeto
    if ltSState == 1
        anyLong := false
    if ltLState == 1
        anyShort := false

// ◆ SB final fire flags — consumed by the label, the exit engine and the alert
sbFireL = anyLong  and sigMCLong  and sbSigLong
sbFireS = anyShort and sigMCShort and sbSigShort

// stamp cooldowns from FINAL signal state only
if sigCRTLong and anyLong
    lastCrtLongBar := bar_index
if sigCRTShort and anyShort
    lastCrtShortBar := bar_index
if sigTBSLong and anyLong
    lastTbsLongBar := bar_index
if sigTBSShort and anyShort
    lastTbsShortBar := bar_index
if anyLong
    lastLongSigBar := bar_index
if anyShort
    lastShortSigBar := bar_index

// ═════════════════════════════════════════════════════════════════ MODULE 19
// TP / SL — structural stop + TP ladder + ACTIVE trade tracking (validation)
// ═════════════════════════════════════════════════════════════════════════
var float trdEntry = na
var float trdSL    = na
var float trdTP1   = na
var float trdTP2   = na
var float trdTP3   = na
var int   trdDir   = 0
var bool  trdTp1Hit = false
// ◆ SB-6 trade state: only an SB-tagged trade uses the extended exit engine
var bool  trdTp2Hit = false
var bool  trdIsSb   = false
var int   trdSbWin  = 0     // the window whose close is this trade's time stop
var int   trdShiftG = 0     // grade of the shift that produced the entry
var float trdLiqTgt = na    // the planned liquidity target (TP3)
bool sbTimeExitEvt   = false
bool sbStructExitEvt = false
bool sbBeEvt         = false

// nearest pool at or beyond `ref` — used to pick the SB liquidity target
f_nearAbove(float best, float cand, float ref) => not na(cand) and cand >= ref and (na(best) or cand < best) ? cand : best
f_nearBelow(float best, float cand, float ref) => not na(cand) and cand <= ref and (na(best) or cand > best) ? cand : best
// validation counters (ARCH-10 — mechanical counts, NOT statistics)
var int cntA      = 0
var int cntAp     = 0
var int cntLong   = 0
var int cntShort  = 0
var int cntTP1    = 0
var int cntTP2    = 0
var int cntSL     = 0
var int cntTP3      = 0
var int cntSbEnt    = 0
var int cntSbTime   = 0
var int cntSbStruct = 0

// resolve the ACTIVE trade before a new one can replace it
// v3.1: a NON-SB trade keeps the v3 path and the v3 counter semantics exactly
// (closed at TP2). An SB-tagged trade runs the SB-6 exit engine instead:
//   TP1  → stop to break-even + sbBeBuf ATR
//   TP2  → stay in, trail behind the external protected swing
//   TP3  → the liquidity target (nearest opposing pool that pays ≥ TP2)
//   time → window close with TP1 still unhit: full exit, or reduce and put
//          the remainder at break-even
//   struct → opposing MSS, or an opposing CHoCH/CISD when the entry shift was
//          only CHoCH/CISD grade (equal or higher quality, never lower)
if confirmed and trdDir != 0
    if not trdIsSb
        if trdDir == 1
            if low <= trdSL
                cntSL += 1
                trdDir := 0
            else if not trdTp1Hit and high >= trdTP1
                trdTp1Hit := true
                cntTP1 += 1
            else if high >= trdTP2
                cntTP2 += 1
                trdDir := 0
        else
            if high >= trdSL
                cntSL += 1
                trdDir := 0
            else if not trdTp1Hit and low <= trdTP1
                trdTp1Hit := true
                cntTP1 += 1
            else if low <= trdTP2
                cntTP2 += 1
                trdDir := 0
    else
        bool sbWEnd = trdSbWin == 1 ? sbEndLOEvt : trdSbWin == 2 ? sbEndAMEvt : trdSbWin == 3 ? sbEndPMEvt : false
        if trdDir == 1
            if low <= trdSL
                cntSL += 1
                trdDir := 0
            else
                if not trdTp1Hit and high >= trdTP1
                    trdTp1Hit := true
                    cntTP1 += 1
                    // ◆ SB-14: ADAPTIVE break-even buffer — the LARGER of the
                    // fixed ATR buffer and SB_BE_TP1_PCT of the TP1 distance.
                    // On a wide-stop trade a flat 0.15 ATR left almost the
                    // whole first leg exposed; this scales with the trade.
                    float beBufL = math.max(sbBeBuf * atr, SB_BE_TP1_PCT * math.abs(trdTP1 - trdEntry))
                    trdSL := math.max(trdSL, trdEntry + beBufL)
                    sbBeEvt := true
                if trdTp1Hit and not trdTp2Hit and high >= trdTP2
                    trdTp2Hit := true
                    cntTP2 += 1
                if trdTp2Hit and sbTrail and not na(protLowE)
                    // ◆ SB-14: trail behind the protected swing, but NEVER
                    // through the last TWO bars' low — that is the immediate
                    // structure the runner needs to breathe against. Still
                    // never through current price either (an already violated
                    // stop would fake a fill on the next bar).
                    float capL = math.min(math.min(low, low[1]), close) - SL_PAD_ATR * atr
                    trdSL := math.max(trdSL, math.min(protLowE - SL_PAD_ATR * atr, capL))
                if trdDir == 1 and high >= trdTP3
                    cntTP3 += 1
                    trdDir := 0
                if trdDir == 1 and sbStructEx and (mssDn or (trdShiftG <= 1 and (cisdDn or chochDn or eChochDn)))
                    cntSbStruct += 1
                    sbStructExitEvt := true
                    trdDir := 0
                if trdDir == 1 and sbTimeExit and sbWEnd and not trdTp1Hit
                    cntSbTime += 1
                    sbTimeExitEvt := true
                    if sbTimeMode == "Full"
                        trdDir := 0
                    else
                        trdSL := math.max(trdSL, trdEntry)
        else
            if high >= trdSL
                cntSL += 1
                trdDir := 0
            else
                if not trdTp1Hit and low <= trdTP1
                    trdTp1Hit := true
                    cntTP1 += 1
                    // ◆ SB-14: adaptive break-even buffer (see the long side)
                    float beBufS = math.max(sbBeBuf * atr, SB_BE_TP1_PCT * math.abs(trdEntry - trdTP1))
                    trdSL := math.min(trdSL, trdEntry - beBufS)
                    sbBeEvt := true
                if trdTp1Hit and not trdTp2Hit and low <= trdTP2
                    trdTp2Hit := true
                    cntTP2 += 1
                if trdTp2Hit and sbTrail and not na(protHighE)
                    // ◆ SB-14: never trail through the last two bars' high
                    float capS = math.max(math.max(high, high[1]), close) + SL_PAD_ATR * atr
                    trdSL := math.min(trdSL, math.max(protHighE + SL_PAD_ATR * atr, capS))
                if trdDir == -1 and low <= trdTP3
                    cntTP3 += 1
                    trdDir := 0
                if trdDir == -1 and sbStructEx and (mssUp or (trdShiftG <= 1 and (cisdUp or chochUp or eChochUp)))
                    cntSbStruct += 1
                    sbStructExitEvt := true
                    trdDir := 0
                if trdDir == -1 and sbTimeExit and sbWEnd and not trdTp1Hit
                    cntSbTime += 1
                    sbTimeExitEvt := true
                    if sbTimeMode == "Full"
                        trdDir := 0
                    else
                        trdSL := math.min(trdSL, trdEntry)
// ◆ SB-12: keep the MODULE 10 dealing-range lock alive while ANY SB narrative
// is still live — an armed SB setup or a running SB trade. Assigned
// unconditionally: MODULE 10 reads it at the top of the NEXT bar.
sbSetLive := (sbTagL != 0 and suL >= 1) or (sbTagS != 0 and suS >= 1) or (trdDir != 0 and trdIsSb)

if confirmed and suL == 4 and trdDir == 0 and not sigMCLong
    suL := 0        // machine ACTIVE state ends with the tracked trade
if confirmed and suS == 4 and trdDir == 0 and not sigMCShort
    suS := 0

float entryP = na
float slP    = na
float tp1P   = na
float tp2P   = na
float tp3P   = na
if (anyLong or anyShort) and confirmed
    entryP := close
    if anyLong
        slP := not na(sigSLLong) and sigSLLong < entryP ? sigSLLong : math.min(low, low[1]) - SL_PAD_ATR * atr
    else
        slP := not na(sigSLShort) and sigSLShort > entryP ? sigSLShort : math.max(high, high[1]) + SL_PAD_ATR * atr
    float risk = math.abs(entryP - slP)
    tp1P := anyLong ? entryP + risk : entryP - risk
    tp2P := anyLong ? entryP + rrMult * risk : entryP - rrMult * risk
    tp3P := anyLong ? entryP + 3 * risk : entryP - 3 * risk
    if tpMode == "Liquidity target"
        float tgt = na
        if anyLong
            if not na(nsBot) and nsBot > entryP
                tgt := nsBot
            else if not na(lastRes) and lastRes > entryP
                tgt := lastRes
            if not na(tgt) and tgt - entryP >= rrMult * risk
                tp3P := tgt
        else
            if not na(nbTop) and nbTop < entryP
                tgt := nbTop
            else if not na(lastSup) and lastSup < entryP
                tgt := lastSup
            if not na(tgt) and entryP - tgt >= rrMult * risk
                tp3P := tgt
    // ◆ SB-6 liquidity target: the NEAREST opposing pool that still pays at
    // least TP2 — an unspent opposite POI, a prior session high/low, or PDH/PDL.
    // If nothing qualifies, TP3 stays on the 3R (or tpMode) value.
    trdLiqTgt := na
    if sbLiqTgt and (sbFireL or sbFireS)
        float need = anyLong ? entryP + rrMult * risk : entryP - rrMult * risk
        float lt = na
        if anyLong
            lt := f_nearAbove(lt, nsBot,     need)
            lt := f_nearAbove(lt, asiaHiLvl, need)
            lt := f_nearAbove(lt, lonHiLvl,  need)
            lt := f_nearAbove(lt, nyHiLvl,   need)
            lt := f_nearAbove(lt, pdh,       need)
