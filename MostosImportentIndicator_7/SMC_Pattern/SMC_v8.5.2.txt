// ── L0 CANDLE ANATOMY — once per bar (R1) ────────────────────────────────────
f_candle(cf, aV, rLo, rHi) =>
    k = Cndl.new()
    k.body   := math.abs(close - open)
    k.rng    := high - low
    k.upWick := high - math.max(close, open)
    k.dnWick := math.min(close, open) - low
    if k.rng > 0
        k.bodyRatio   := k.body   / k.rng
        k.upWickRatio := k.upWick / k.rng
        k.dnWickRatio := k.dnWick / k.rng
    k.isBull     := close > open
    k.isBear     := close < open
    k.largeRange := k.rng > aV * cf.largeRangeMult
    if bar_index > 1
        pBody = math.abs(close[1] - open[1])
        k.pinBull := k.dnWick >= cf.pinWickMult * k.body and k.dnWickRatio >= cf.pinWickRatioMin and k.upWick <= k.body
        k.pinBear := k.upWick >= cf.pinWickMult * k.body and k.upWickRatio >= cf.pinWickRatioMin and k.dnWick <= k.body
        k.doji    := k.bodyRatio <= cf.dojiBodyMax
        k.engulfBull := k.isBull and close >= open[1] and open <= close[1] and close[1] < open[1]
        k.engulfBear := k.isBear and close <= open[1] and open >= close[1] and close[1] > open[1]
        k.insideBar  := high <= high[1] and low >= low[1]
        k.harami     := high < high[1] and low > low[1] and k.body < pBody * cf.haramiBodyMax
        k.longWickRej := (k.upWickRatio >= cf.longWickRatio or k.dnWickRatio >= cf.longWickRatio) and k.largeRange
        k.starBear := k.pinBear and math.abs(high - rHi) <= aV * cf.starProxATR
        k.starBull := k.pinBull and math.abs(low - rLo) <= aV * cf.starProxATR
        okBull = cf.ifcAnyColor or close > open
        okBear = cf.ifcAnyColor or close < open
        k.ifcBull := low  < rLo and okBull and k.dnWick >= cf.ifcWickMult * k.body
        k.ifcBear := high > rHi and okBear and k.upWick >= cf.ifcWickMult * k.body
        k.manipBull := low  < low[1]  and close > high[1] and close > open and k.dnWickRatio >= cf.manipWickRatio
        k.manipBear := high > high[1] and close < low[1]  and close < open and k.upWickRatio >= cf.manipWickRatio
    k

// S2 (audit B1) — origin of the leg that just broke a swing: the lowest low (bull) / highest high
// (bear) printed AFTER the broken swing and BEFORE this bar. f_moveStrength used the pre-shift
// pivot and f_zones the post-shift one; on a bar that is both a BOS and a pivot confirmation the
// two engines measured two different legs, and neither was the price origin. Returns [idx, price].
f_legOrigin(isBull, fromIdx) =>
    span = math.min(math.max(bar_index - fromIdx, 1), 490)
    bestOff = 1
    bestVal = isBull ? low[1] : high[1]
    if span >= 2
        for k = 2 to span
            v = isBull ? low[k] : high[k]
            if (isBull and v < bestVal) or (not isBull and v > bestVal)
                bestVal := v
                bestOff := k
    [bar_index - bestOff, bestVal]

// v8.5.1 S2 — highest high (isHi) / lowest low from fromIdx through this bar, ≤ 491 bars. Called only when
// the PD anchor moves; the guard matters because `for k = 1 to 0` would run k = 1, 0.
f_legExt(isHi, fromIdx) =>
    span = math.min(math.max(bar_index - fromIdx, 0), 490)
    v = isHi ? high : low
    if span > 0
        for k = 1 to span
            v := isHi ? math.max(v, high[k]) : math.min(v, low[k])
    v

// v8.5.2 E-4 — (declared here: used by the M74 pre-conditions, long before the trade engine) CE of the unmitigated FVG of the current leg (idx ≥ fromIdx) nearest to px on the entry side
f_legFvg(cx, isLong, fromIdx, px) =>
    best = float(na)
    n = array.size(cx.fvgs)
    if n > 0
        for i = 0 to n - 1
            g = array.get(cx.fvgs, i)
            if g.mitigated or g.isBull != isLong or g.idx < fromIdx
                continue
            ce = (g.top + g.bottom) * 0.5
            if (isLong and ce < px and (na(best) or ce > best)) or (not isLong and ce > px and (na(best) or ce < best))
                best := ce
    best

// S4 — arm a break that lacks displacement; a second close beyond `lvl` within pendBars promotes it
f_pend(sx, dir, kind, lvl, fromIdx) =>
    sx.stPendDir  := dir
    sx.stPendKind := kind
    sx.stPendBar  := bar_index
    sx.stPendLvl  := lvl
    sx.stPendFrom := fromIdx

// ── L1 STRUCTURE ENGINE — once per bar (R1) ──────────────────────────────────
// S4 (audit B2 / B3 / RC1). v8.2 emitted a BOS on ANY close through the last pivot, flipped the trend
// on it, and tested CHoCH afterwards — so majorHigh ≥ curHigh made CHoCH unreachable and every
// corrective break read as a trend change. Now:
//   BOS       = close through the last pivot IN the trend direction (or from RANGE), with displacement
//   CHoCH     = close through the PROTECTED swing (majorHigh / majorLow) AGAINST the trend, with displacement
//   internal  = counter-trend close through a pivot that is not the protected swing → label only
//   weak      = either of the first two without displacement → pending; promoted by a follow-through
//               close within pendBars, otherwise dropped
//   RANGE     = two consecutive breaks that closed back through their level, or the old bar count
// The protected swing is the price origin of the leg that made the break (f_legOrigin), stored in
// majorLow (bull trend) / majorHigh (bear trend) so every downstream reader keeps its name.
f_structure(cf, cx, sx, swH, swL, aN, dispOk, dDist, pendBars, failMax) =>
    s = Struct.new()
    pIdx = bar_index - cf.swingLen
    if not na(swH)
        s.swingHigh := true
        sx.hiType   := na(cx.curHigh) ? "" : swH > cx.curHigh ? "HH" : "LH"
    if not na(swL)
        s.swingLow := true
        sx.loType  := na(cx.curLow) ? "" : swL < cx.curLow ? "LL" : "HL"
    // failed follow-through: the last break (of any kind) closed back through its level
    if sx.stBrkDir != 0 and bar_index - sx.stBrkBar >= 1
        if bar_index - sx.stBrkBar <= cf.failedCloseBars
            if (sx.stBrkDir == 1 and close < sx.stBrkLvl) or (sx.stBrkDir == -1 and close > sx.stBrkLvl)
                sx.stFail   := sx.stFail + 1
                sx.stBrkDir := 0
                if sx.stFail >= failMax
                    cx.trend := "RANGE"
        else
            // v8.5.0 W12 — a break that survived its window is a success: the counter is "consecutive" fails
            sx.stBrkDir := 0
            sx.stFail   := 0
    brkUp = not na(cx.curHigh) and not cx.hiBroken and close > cx.curHigh
    brkDn = not na(cx.curLow)  and not cx.loBroken and close < cx.curLow
    s.bullSweep := not na(cx.curHigh) and not cx.hiBroken and high > cx.curHigh and close <= cx.curHigh
    s.bearSweep := not na(cx.curLow)  and not cx.loBroken and low  < cx.curLow  and close >= cx.curLow
    // §M1 — a gap or outside bar can close beyond BOTH stored extremes. Keep the side displaced further.
    if brkUp and brkDn
        if (close - cx.curHigh) >= (cx.curLow - close)
            brkDn := false
        else
            brkUp := false
    if brkUp
        cx.hiBroken := true
    if brkDn
        cx.loBroken := true
    // pending weak break: promote on a follow-through close, expire after pendBars
    promUp = false
    promDn = false
    if sx.stPendDir != 0
        if bar_index - sx.stPendBar > pendBars
            sx.stPendDir := 0
        else if sx.stPendDir == 1 and close > sx.stPendLvl
            promUp := true
            if sx.stPendKind == 2
                s.bullCHoCH := true
                s.chLvl     := sx.stPendLvl
                s.chIdx     := sx.stPendFrom
            else
                s.bullBOS := true
            sx.stPendDir := 0
        else if sx.stPendDir == -1 and close < sx.stPendLvl
            promDn := true
            if sx.stPendKind == 2
                s.bearCHoCH := true
                s.chLvl     := sx.stPendLvl
                s.chIdx     := sx.stPendFrom
            else
                s.bearBOS := true
            sx.stPendDir := 0
    // R2 (gold review) — the protected swing is the leg ORIGIN, which after a long leg sits far away;
    // a two-week decline of LH + LL never reached it, so every break stayed "internal" and the trend
    // read BULLISH → RANGE. ICT rule: in a bull trend a LOWER high followed by a close through the
    // last pivot low IS the change of character. Same for HL + close through the last pivot high.
    chUpMaj = cx.trend == "BEARISH" and not na(cx.majorHigh) and close > cx.majorHigh
    chDnMaj = cx.trend == "BULLISH" and not na(cx.majorLow)  and close < cx.majorLow
    chUp = chUpMaj or (cx.trend == "BEARISH" and brkUp and sx.loType == "HL")
    chDn = chDnMaj or (cx.trend == "BULLISH" and brkDn and sx.hiType == "LH")
    // v8.4.1 H1 — the level the CHoCH actually closed through: the protected swing, or in the R2 form the
    // last pivot. Recording majorHigh for the R2 form left the pending break / failed-follow-through
    // counter / CHoCH line pointing at a level far from price (or na).
    chUpLvl = chUpMaj ? cx.majorHigh    : cx.curHigh
    chUpIdx = chUpMaj ? cx.majorHighIdx : cx.curHighIdx
    chDnLvl = chDnMaj ? cx.majorLow     : cx.curLow
    chDnIdx = chDnMaj ? cx.majorLowIdx  : cx.curLowIdx
    // R1 — displacement = strong candle OR a close that clears the level by dDist
    dBrkUp = dispOk or (not na(cx.curHigh)   and close - cx.curHigh   >= dDist)
    dChUp  = dispOk or (not na(cx.majorHigh) and close - cx.majorHigh >= dDist) or dBrkUp
    dBrkDn = dispOk or (not na(cx.curLow)    and cx.curLow  - close   >= dDist)
    dChDn  = dispOk or (not na(cx.majorLow)  and cx.majorLow - close  >= dDist) or dBrkDn
    evtAny = s.bullBOS or s.bearBOS or s.bullCHoCH or s.bearCHoCH
    // bull side
    if (brkUp or chUp) and not evtAny
        if chUp
            if dChUp
                s.bullCHoCH := true
                s.chLvl     := chUpLvl
                s.chIdx     := chUpIdx
            else
                s.weakBullBOS := true
                f_pend(sx, 1, 2, chUpLvl, chUpIdx)
        else if cx.trend == "BEARISH"
            s.internalBullBOS := true
        else if dBrkUp
            s.bullBOS := true
        else
            s.weakBullBOS := true
            f_pend(sx, 1, 1, cx.curHigh, cx.curHighIdx)
    evtAny := s.bullBOS or s.bearBOS or s.bullCHoCH or s.bearCHoCH
    // bear side
    if (brkDn or chDn) and not evtAny and not s.weakBullBOS and not s.internalBullBOS
        if chDn
            if dChDn
                s.bearCHoCH := true
                s.chLvl     := chDnLvl
                s.chIdx     := chDnIdx
            else
                s.weakBearBOS := true
                f_pend(sx, -1, 2, chDnLvl, chDnIdx)
        else if cx.trend == "BULLISH"
            s.internalBearBOS := true
        else if dBrkDn
            s.bearBOS := true
        else
            s.weakBearBOS := true
            f_pend(sx, -1, 1, cx.curLow, cx.curLowIdx)
    // state changes — only a displaced (or promoted) BOS / CHoCH moves the trend and the protected swing
    if s.bullBOS or s.bullCHoCH
        fromI = promUp ? sx.stPendFrom : s.bullCHoCH ? s.chIdx : cx.curHighIdx
        [oi, op] = f_legOrigin(true, fromI)
        cx.legIdx       := oi
        cx.legPx        := op
        cx.majorLow     := op
        cx.majorLowIdx  := oi
        cx.trend        := "BULLISH"
        cx.barsSinceEvt := 0
        sx.stFail       := 0
        sx.stBrkDir     := 1
        sx.stBrkLvl     := s.bullCHoCH ? s.chLvl : cx.curHigh
        sx.stBrkBar     := bar_index
        if sx.stPendDir == -1                   // v8.5.0 D8 — a stale opposite weak break cannot promote now
            sx.stPendDir := 0
        if s.bullCHoCH
            // §M2 — upside is taken, a fresh downside break is meaningful again
            cx.hiBroken := true
            cx.loBroken := false
    if s.bearBOS or s.bearCHoCH
        fromI = promDn ? sx.stPendFrom : s.bearCHoCH ? s.chIdx : cx.curLowIdx
        [oi, op] = f_legOrigin(false, fromI)
        cx.legIdx       := oi
        cx.legPx        := op
        cx.majorHigh    := op
        cx.majorHighIdx := oi
        cx.trend        := "BEARISH"
        cx.barsSinceEvt := 0
        sx.stFail       := 0
        sx.stBrkDir     := -1
        sx.stBrkLvl     := s.bearCHoCH ? s.chLvl : cx.curLow
        sx.stBrkBar     := bar_index
        if sx.stPendDir == 1
            sx.stPendDir := 0
        if s.bearCHoCH
            cx.loBroken := true
            cx.hiBroken := false
    // weak / internal breaks: measure the leg (for mv). Only a WEAK (trend-aligned) break feeds the
    // failure counter — R1: a counter-trend internal break failing is what a trend looks like, and
    // counting it sent 5m gold / NAS into RANGE inside clean trends.
    if s.weakBullBOS or s.internalBullBOS
        [oi, op] = f_legOrigin(true, cx.curHighIdx)
        cx.legIdx   := oi
        cx.legPx    := op
        if s.weakBullBOS
            sx.stBrkDir := 1
            sx.stBrkLvl := sx.stPendDir == 1 ? sx.stPendLvl : cx.curHigh   // v8.5.0 D9 — the level actually broken
            sx.stBrkBar := bar_index
    if s.weakBearBOS or s.internalBearBOS
        [oi, op] = f_legOrigin(false, cx.curLowIdx)
        cx.legIdx   := oi
        cx.legPx    := op
        if s.weakBearBOS
            sx.stBrkDir := -1
            sx.stBrkLvl := sx.stPendDir == -1 ? sx.stPendLvl : cx.curLow
            sx.stBrkBar := bar_index
    if not (s.bullBOS or s.bearBOS or s.bullCHoCH or s.bearCHoCH)
        cx.barsSinceEvt := cx.barsSinceEvt + 1
        if cx.barsSinceEvt > cf.rangeBarsThresh
            cx.trend := "RANGE"
    if not na(swH)
        cx.prevHigh    := cx.curHigh
        cx.prevHighIdx := cx.curHighIdx
        cx.curHigh     := swH
        cx.curHighIdx  := pIdx
        cx.hiBroken    := false
    if not na(swL)
        cx.prevLow    := cx.curLow
        cx.prevLowIdx := cx.curLowIdx
        cx.curLow     := swL
        cx.curLowIdx  := pIdx
        cx.loBroken   := false
    // §3.5 pullback validity — highest-high / lowest-low candle tracking (tie → latest)
    // the maker must stay the genuine extreme-maker candle: overwriting it with the pullback candle
    // made `low < hiMakerLow` true again on the next bar of the same sequence, so pbValidUp fired on
    // EVERY bar of a move (flooding PB labels and resetting the leg counters every bar). The armed
    // flag makes it one-shot per extreme; a new extreme re-arms it.
    if na(cx.hiMakerHigh) or high >= cx.hiMakerHigh
        cx.hiMakerHigh := high
        cx.hiMakerLow  := low
        cx.pbUpArmed   := true
    else if cx.pbUpArmed and low < cx.hiMakerLow
        s.pbValidUp    := true
        cx.pbUpArmed   := false
    if na(cx.loMakerLow) or low <= cx.loMakerLow
        cx.loMakerLow  := low
        cx.loMakerHigh := high
        cx.pbDnArmed   := true
    else if cx.pbDnArmed and high > cx.loMakerHigh
        s.pbValidDn    := true
        cx.pbDnArmed   := false
    // §C1 — the makers only ever moved outwards, so after a few hundred bars hiMakerHigh WAS the
    // all-time high and loMakerLow the all-time low. movePhase then reduced to "is close at the
    // all-time extreme", i.e. permanently RETRACEMENT, and the pullback tracker was comparing
    // against a candle from a different market regime. A structural event ends the old leg, so
    // re-anchor both makers on it and re-arm the one-shot pullback flags.
    if s.bullBOS or s.bearBOS or s.bullCHoCH or s.bearCHoCH
        cx.hiMakerHigh := high
        cx.hiMakerLow  := low
        cx.pbUpArmed   := true
        cx.loMakerLow  := low
        cx.loMakerHigh := high
        cx.pbDnArmed   := true
    // §3.6 CISD · S10 (audit T11) — ICT definition: the reference is the OPEN of the first candle of
    // the same-colour series that delivered the leg extreme; the CISD is a close through it, and it
    // only means something after liquidity was taken. v8.2 used the open of the first green candle
    // after the last pullback, which was a leg invalidation that fired late in trends and early on
    // every pullback.
    if close > open and not (close[1] > open[1])
        cx.cisdUpOpen := open                       // running up-series start
        cx.cisdUpBar  := bar_index
    if close < open and not (close[1] < open[1])
        cx.cisdDnOpen := open
        cx.cisdDnBar  := bar_index
    if high >= cx.hiMakerHigh and not na(cx.cisdUpOpen)      // this bar made the leg high
        sx.cisdUpRef    := cx.cisdUpOpen
        sx.cisdUpRefBar := cx.cisdUpBar
        sx.cisdUpUsed   := false
    if low <= cx.loMakerLow and not na(cx.cisdDnOpen)
        sx.cisdDnRef    := cx.cisdDnOpen
        sx.cisdDnRefBar := cx.cisdDnBar
        sx.cisdDnUsed   := false
    swOkUp = not cf.cisdNeedSweep or (cx.sweptHighBar >= 0 and bar_index - cx.sweptHighBar <= cf.armBars)
    swOkDn = not cf.cisdNeedSweep or (cx.sweptLowBar  >= 0 and bar_index - cx.sweptLowBar  <= cf.armBars)
    s.cisdBear := not sx.cisdUpUsed and not na(sx.cisdUpRef) and close < sx.cisdUpRef and swOkUp
    s.cisdBull := not sx.cisdDnUsed and not na(sx.cisdDnRef) and close > sx.cisdDnRef and swOkDn
    if s.cisdBear
        sx.cisdUpUsed := true
    if s.cisdBull
        sx.cisdDnUsed := true
    s.trend     := cx.trend
    // §C1 · §3.8 · S4 — measured against the current leg IN the trend direction: price pushing past
    // the live swing that defines the leg is impulse, anything inside it is retracement.
    s.movePhase := cx.trend == "RANGE" ? "RANGE" :
                   (cx.trend == "BULLISH" and not na(cx.curHigh) and close >= cx.curHigh) or
                   (cx.trend == "BEARISH" and not na(cx.curLow)  and close <= cx.curLow) ? "IMPULSIVE" : "RETRACEMENT"
    s

// ── drawing x-coordinate clamp ──────────────────────────────────────
// xloc.bar_index drawings may not reference a bar older than max_bars_back (500),
// otherwise: "The requested historical offset is beyond the historical buffer's limit".
// Stored pivot / zone indices can outlive that window, so pin them to the edge.
f_xb(x) => math.max(math.min(x, bar_index), bar_index - 495)

// ── L2 FVG ENGINE — once per bar (R1); writes into the shared Feat ───────────
f_fvg(cf, cx, v, fv, aV, drawOn, keepMit) =>
    minH = aV * cf.fvgMinATR
    // §C2 — the third candle of the gap is `close`'s own bar. Reading its live low meant the gap's
    // top moved on every tick and could disappear entirely at the close, and since mv.fvgCount,
    // chochReal, atBullFvg/atBearFvg, TP1 and the confluence score all hang off the FVG list, one
    // repainting box repainted the whole signal. Every other zone engine here is already
    // confirmed-gated; this one was the outlier.
    if barstate.isconfirmed and bar_index > 2
        if low > high[2] and (low - high[2]) >= minH
            g = Fvg.new(top = low, bottom = high[2], idx = bar_index - 1, isBull = true)
            if drawOn
                g.bx := box.new(bar_index - 1, low, bar_index, high[2], border_color = color.new(v.bull, 100), bgcolor = color.new(v.bull, v.boxT), extend = extend.right)
            array.push(cx.fvgs, g)
            fv.newBullFvg := true
        if high < low[2] and (low[2] - high) >= minH
            g = Fvg.new(top = low[2], bottom = high, idx = bar_index - 1, isBull = false)
            if drawOn
                g.bx := box.new(bar_index - 1, low[2], bar_index, high, border_color = color.new(v.bear, 100), bgcolor = color.new(v.bear, v.boxT), extend = extend.right)
            array.push(cx.fvgs, g)
            fv.newBearFvg := true
    n = array.size(cx.fvgs)
    if n > 0
        for i = n - 1 to 0
            g = array.get(cx.fvgs, i)
            if not g.mitigated and g.idx < bar_index - 1
                mid = (g.top + g.bottom) * 0.5
                lvl = cf.fvgMitMode == "TOUCH" ? (g.isBull ? g.top : g.bottom) :
                      cf.fvgMitMode == "50%"   ? mid : (g.isBull ? g.bottom : g.top)
                hit = g.isBull ? low <= lvl : high >= lvl
                if hit
                    g.mitigated := true
                    g.mitIdx    := bar_index
                    if not na(g.bx)
                        box.set_extend(g.bx, extend.none)
                        box.set_right(g.bx, bar_index)
                        if keepMit
                            cCol = g.isBull ? v.bull : v.bear
                            box.set_bgcolor(g.bx, color.new(v.inv, v.boxMitT))
                            box.set_border_color(g.bx, color.new(v.inv, 100))
                        else
                            box.delete(g.bx)
            if g.mitigated and bar_index - g.mitIdx > cf.fvgMaxAge
                if not na(g.bx)
                    box.delete(g.bx)
                array.remove(cx.fvgs, i)
    while array.size(cx.fvgs) > cf.maxFvg
        old = array.shift(cx.fvgs)
        if not na(old.bx)
            box.delete(old.bx)

// nearest FVG of a direction whose middle candle sits in [fromIdx, toIdx] → bottom (bull) / top (bear), na if none
f_fvgNear(cx, isBull, fromIdx, toIdx) =>
    res = float(na)
    n = array.size(cx.fvgs)
    if n > 0
        for i = 0 to n - 1
            g = array.get(cx.fvgs, i)
            if g.isBull == isBull and g.idx >= fromIdx and g.idx <= toIdx
                res := isBull ? g.bottom : g.top
                break
    res

// ── L2 MOVE STRENGTH — EVENT-DRIVEN ONLY (§7): BOS, zone tap, CHoCH ──────────
// v8.5.0 D6 — `isBull` = the leg direction: only gaps of that direction count as its imbalance
f_moveStrength(cf, cx, fromBar, toBar, aV, isBull) =>
    m = Move.new()
    fb = math.max(math.max(fromBar, toBar - cf.maxMoveBars + 1), 0)
    cnt = toBar - fb + 1
    if cnt > 0 and aV > 0
        bodySum = 0.0
        rangeSum = 0.0
        flips = 0
        prevBull = false
        for b = fb to toBar
            off = bar_index - b
            if off >= 0 and off < 500
                bodySum  += math.abs(close[off] - open[off])
                rangeSum += high[off] - low[off]
                isB = close[off] > open[off]
                if b > fb and isB != prevBull
                    flips += 1
                prevBull := isB
        k = array.size(cx.fvgs)
        if k > 0
            for i = 0 to k - 1
                g = array.get(cx.fvgs, i)
                if g.isBull == isBull and g.idx >= fb and g.idx <= toBar and not g.mitigated
                    m.fvgCount := m.fvgCount + 1
        m.avgBody := bodySum / cnt
        m.bodyEff := rangeSum > 0 ? bodySum / rangeSum : 0.0
        m.mixed   := cnt > 1 ? flips / (cnt - 1) : 0.0
        offTo   = math.min(math.max(bar_index - toBar, 0), 499)
        offFrom = math.min(math.max(bar_index - fb, 0), 499)
        netMove = math.abs(close[offTo] - open[offFrom])
        m.speed  := netMove / cnt / aV
        m.strong := m.fvgCount >= cf.minFVGForStrong and m.bodyEff >= cf.strongBodyEff and
                     m.mixed <= cf.strongMixMax and m.speed >= cf.strongSpeedMin
    m

// ── L3 LIQUIDITY — helpers ───────────────────────────────────────────────────
// v8.5.0 W7 — POOL = `pooled` (EQH/EQL merge of two pivots, or ≥ 3 touches), not "one retest wick"
f_lvlColor(v, L) =>
    L.pooled ? v.pool : L.kind == "PDH" or L.kind == "PDL" or L.kind == "PWH" or L.kind == "PWL" ? v.neut : L.isHigh ? v.bear : v.bull

f_lvlText(L) =>
    (L.pooled ? "POOL·" : "") + L.kind + (L.ext ? "·ext" : "")

f_resolve(v, L, res) =>
    L.done   := true
    L.result := res
    if not na(L.ln)
        line.set_extend(L.ln, extend.none)
        line.set_x2(L.ln, bar_index)
        line.set_style(L.ln, res == "SWEEP" ? line.style_dotted : line.style_solid)
        line.set_color(L.ln, color.new(res == "SWEEP" ? f_lvlColor(v, L) : v.neut, math.min(v.lineT + 30, 100)))

f_addLevel(cf, cx, v, aV, price, idx, isHigh, kind, ext, drawOn) =>
    merged = false
    n = array.size(cx.lv)
    if n > 0 and (kind == "BSL" or kind == "SSL")
        for i = n - 1 to 0
            L = array.get(cx.lv, i)
            // only merge into another swing-derived level: renaming a PDH/PWH to EQH would strand it,
            // because the daily/weekly rollover expires levels by kind and would no longer match it
            famOk = L.kind == "BSL" or L.kind == "SSL" or L.kind == "EQH" or L.kind == "EQL"
            if not L.done and famOk and L.isHigh == isHigh and math.abs(L.price - price) <= aV * cf.eqTolATR
                L.kind      := isHigh ? "EQH" : "EQL"
                L.touches   := L.touches + 1
                L.lastTouch := bar_index
                L.price     := (L.price + price) * 0.5
                L.pooled    := true
                if not na(L.ln)
                    line.set_y1(L.ln, L.price)
                    line.set_y2(L.ln, L.price)
                    line.set_width(L.ln, 2)
                    line.set_style(L.ln, line.style_dashed)
                    line.set_color(L.ln, color.new(f_lvlColor(v, L), v.lineT))
                if not na(L.lb)
                    label.set_text(L.lb, f_lvlText(L))
                    label.set_textcolor(L.lb, f_lvlColor(v, L))
                else if drawOn
                    L.lb := label.new(f_xb(idx), L.price, f_lvlText(L), style = label.style_none, textcolor = f_lvlColor(v, L), size = v.zsz)
                merged := true
                break
    if not merged
        L = Liq.new(price = price, idx = idx, isHigh = isHigh, kind = kind, ext = ext, lastTouch = bar_index)
        if drawOn
            L.ln := line.new(f_xb(idx), price, bar_index, price, color = color.new(f_lvlColor(v, L), v.lineT), width = 1, style = line.style_dotted, extend = extend.right)
        array.push(cx.lv, L)
    merged

// ── L3 LIQUIDITY ENGINE — once per bar (R1); writes into the shared Feat ─────
f_liq(cf, cx, v, fv, aV, swH, swL, eHi, eLo, pdh, pdl, pwh, pwl, nDay, nWeek, drawOn) =>
    pIdx = bar_index - cf.swingLen
    rng  = high - low
    bodyR = rng > 0 ? math.abs(close - open) / rng : 0.0
    if not na(swH)
        f_addLevel(cf, cx, v, aV, swH, pIdx, true,  "BSL", swH >= eHi, drawOn)
    if not na(swL)
        f_addLevel(cf, cx, v, aV, swL, pIdx, false, "SSL", swL <= eLo, drawOn)
    if nDay and not na(pdh)
        if array.size(cx.lv) > 0
            for i = array.size(cx.lv) - 1 to 0
                L = array.get(cx.lv, i)
                if not L.done and (L.kind == "PDH" or L.kind == "PDL")
                    f_resolve(v, L, "EXPIRED")
        f_addLevel(cf, cx, v, aV, pdh, bar_index, true,  "PDH", false, drawOn)
        f_addLevel(cf, cx, v, aV, pdl, bar_index, false, "PDL", false, drawOn)
    if nWeek and not na(pwh)
        if array.size(cx.lv) > 0
            for i = array.size(cx.lv) - 1 to 0
                L = array.get(cx.lv, i)
                if not L.done and (L.kind == "PWH" or L.kind == "PWL")
                    f_resolve(v, L, "EXPIRED")
        f_addLevel(cf, cx, v, aV, pwh, bar_index, true,  "PWH", false, drawOn)
        f_addLevel(cf, cx, v, aV, pwl, bar_index, false, "PWL", false, drawOn)
    n = array.size(cx.lv)
    scanD = aV * cf.nearZoneATR
    if n > 0
        for i = n - 1 to 0
            L = array.get(cx.lv, i)
            if L.done
                if bar_index - L.lastTouch > cf.extLookback * 4
                    if not na(L.ln)
                        line.delete(L.ln)
                    if not na(L.lb)
                        label.delete(L.lb)
                    array.remove(cx.lv, i)
                continue
            if L.idx >= bar_index
                continue
            tol   = aV * cf.touchTolATR
            minGo = aV * cf.runCloseMinATR
            if L.isHigh
                if L.pend > 0
                    L.wickX := math.max(L.wickX, high)          // v8.5.0 D2
                    if close < L.price
                        f_resolve(v, L, "SWEEP")
                    else
                        L.pend := L.pend - 1
                        if L.pend == 0
                            f_resolve(v, L, "RUN")
                else if high > L.price
                    if close <= L.price
                        f_resolve(v, L, "SWEEP")
                    else if bodyR >= cf.runBodyMin and (close - L.price) >= minGo
                        f_resolve(v, L, "RUN")
                    else
                        L.pend  := cf.failedCloseBars
                        L.wickX := high
                else if high >= L.price - tol and bar_index - L.lastTouch > cf.touchCooldown
                    L.touches   := L.touches + 1
                    L.lastTouch := bar_index
            else
                if L.pend > 0
                    L.wickX := math.min(L.wickX, low)
                    if close > L.price
                        f_resolve(v, L, "SWEEP")
                    else
                        L.pend := L.pend - 1
                        if L.pend == 0
                            f_resolve(v, L, "RUN")
                else if low < L.price
                    if close >= L.price
                        f_resolve(v, L, "SWEEP")
                    else if bodyR >= cf.runBodyMin and (L.price - close) >= minGo
                        f_resolve(v, L, "RUN")
                    else
                        L.pend  := cf.failedCloseBars
                        L.wickX := low
                else if low <= L.price + tol and bar_index - L.lastTouch > cf.touchCooldown
                    L.touches   := L.touches + 1
                    L.lastTouch := bar_index
            if L.touches >= 3 and not L.pooled                 // v8.5.0 W7
                L.pooled := true
                if not na(L.ln)
                    line.set_width(L.ln, 2)
                    line.set_style(L.ln, line.style_dashed)
                    line.set_color(L.ln, color.new(v.pool, v.lineT))
                if not na(L.lb)
                    label.set_text(L.lb, f_lvlText(L))
                    label.set_textcolor(L.lb, v.pool)
                else if drawOn
                    L.lb := label.new(f_xb(L.idx), L.price, f_lvlText(L), style = label.style_none, textcolor = v.pool, size = v.zsz)
            if L.done
                if L.result == "SWEEP"
                    // S5 (audit T3 / §9) — only a QUALIFIED level (pool, PDH/PDL, PWH/PWL, SESH/SESL,
                    // optionally the extLookback extreme) arms the sweep-based chain (HP OB, M41, M45,
                    // M54, CISD reliability, the ✕ marker, f_conf). A plain pivot sweep is internal.
                    qual = L.pooled or L.kind == "PDH" or L.kind == "PDL" or L.kind == "PWH" or L.kind == "PWL" or L.kind == "SESH" or L.kind == "SESL" or (cf.sweepExtOk and L.ext)
                    if L.isHigh
                        fv.sweptHigh := true
                        fv.sweptExtHi := math.max(nz(L.wickX, high), high)   // v8.5.0 D2
                        if qual
                            fv.sweptHighQ   := true
                            cx.sweptHighBar := bar_index
                    else
                        fv.sweptLow := true
                        fv.sweptExtLo := math.min(nz(L.wickX, low), low)
                        if qual
                            fv.sweptLowQ   := true
                            cx.sweptLowBar := bar_index
                    fv.sweptKind  := f_lvlText(L)
                    if L.isHigh
                        fv.sweptKindHi  := f_lvlText(L)
                        fv.sweptPriceHi := L.price
                    else
                        fv.sweptKindLo  := f_lvlText(L)
                        fv.sweptPriceLo := L.price
                    fv.sweptPrice := L.price
                    // `ext` (at the extLookback extreme) is the common case on a low timeframe, so
                    // counting it here let nearly every sweep through the "major levels only" filter
                    fv.sweptMajor := fv.sweptMajor or L.pooled or L.kind == "PDH" or L.kind == "PDL" or L.kind == "PWH" or L.kind == "PWL"
                if L.result == "RUN"
                    if L.isHigh
                        fv.runUp := true
                    else
                        fv.runDn := true
                    fv.runPrice := L.price
                    fv.runMajor := fv.runMajor or L.pooled or L.ext or L.kind == "PDH" or L.kind == "PDL" or L.kind == "PWH" or L.kind == "PWL"
            else
                // liquidity targets near price (M09 / global scoring): unresolved levels within scan distance
                if L.isHigh and L.price > close and L.price - close <= scanD
                    fv.bslAbove := fv.bslAbove + 1
                if not L.isHigh and L.price < close and close - L.price <= scanD
                    fv.sslBelow := fv.sslBelow + 1
    // §M4 — eviction used to be array.shift, i.e. "oldest inserted wins the bin". PDH/PDL enter at
    // the day's first bar, so on a busy session the day's own levels were the first thing thrown
    // away. Score each level instead and drop the cheapest: resolved levels go first, then the ones
    // furthest from price; pooled / daily / weekly / session levels are the last to go.
    while array.size(cx.lv) > cf.maxLevels
        wIdx = 0
        for i = 0 to array.size(cx.lv) - 1
            Le = array.get(cx.lv, i)
            if Le.done or (not Le.pooled and (Le.kind == "BSL" or Le.kind == "SSL"))
                wIdx := i
                break
        old = array.get(cx.lv, wIdx)
        if not na(old.ln)
            line.delete(old.ln)
        if not na(old.lb)
            label.delete(old.lb)
        array.remove(cx.lv, wIdx)

// ── L4 ZONE — helpers ────────────────────────────────────────────────────────
f_zColor(v, z) =>
    z.kind == "OB"  ? (z.bull ? v.bull : v.bear) :
     z.kind == "BRK" ? v.brk : z.kind == "RCB" ? v.rcb : z.kind == "RJB" ? v.rjb : z.kind == "QM" ? v.qm :
     z.kl >= 4 ? (z.bull ? v.bull : v.bear) : v.sr

f_zText(z) =>
    t = z.kind == "OB" ? (z.bull ? "OB▲" : "OB▼") + (z.hp ? "★" : "") + (z.obType != "" ? " " + z.obType : "") :
         z.kind == "SR" ? (z.kl >= 4 ? "KL★" : z.kl == 3 ? "KL" : (z.bull ? "S" : "R")) :
         z.kind + (z.bull ? "▲" : "▼")
    t += z.flipped ? " FLIP" : ""
    t += z.sweepReq ? " ⚠SWEEP REQ" : ""
    t += z.weak ? " weak" : ""
    t += not z.confirmed ? " ?" : ""
    t += z.idmSwept ? " IDM✓" : ""
    t += z.prop ? " ⇈PROP" : ""
    t += z.nearSwept ? " ★★" : ""
    t

f_zStyle(v, z) =>
    if not na(z.bx)
        weakZ = z.broken or z.weak or not z.confirmed or (z.kind == "OB" and not z.hp and z.obType != "")
        c = weakZ and (z.broken or z.weak) ? v.inv : f_zColor(v, z)
        box.set_bgcolor(z.bx, color.new(c, z.broken or z.weak ? 90 : z.tapped ? math.min(v.boxT + 6, 100) : v.boxT))
        box.set_border_color(z.bx, color.new(z.sweepReq ? v.inv : c, z.broken ? 80 : 40))
        box.set_border_width(z.bx, z.kl >= 4 ? 3 : z.kl == 3 ? 2 : 1)
        box.set_border_style(z.bx, weakZ or z.kind == "BRK" or z.flipped ? line.style_dashed : line.style_solid)
        box.set_text(z.bx, f_zText(z))
        box.set_text_size(z.bx, v.zsz)
        // the box fill IS colour `c` at boxT transparency — painting the text the same colour makes
        // it vanish. chart.fg_color is white on a dark chart, black on a light one.
        box.set_text_color(z.bx, chart.fg_color)

f_zAdd(cf, cx, v, aV, kind, bull, top, bottom, idx, drawOn) =>
    zt = math.max(top, bottom)
    zb = math.min(top, bottom)
    maxH = aV * cf.zoneMaxATR
    if zt - zb > maxH
        // oversized (flash-wick candle): keep the origin edge, clip the rest
        if bull
            zt := zb + maxH
        else
            zb := zt - maxH
    // v8.2.1 · three zones of the same kind within a few ticks was normal on the review charts:
    // clutter on the chart, and f_conf counted one price as "a zone" once per copy. An overlapping
    // live zone of the same kind and side is refreshed instead of duplicated. Two kinds are exempt:
    // SR, because the key-level block finds its origin by `z.idx == oIdx` and merging would break
    // scoring, and RJB, because the caller resets `confirmed` / `pendFvg` on whatever comes back -
    // merging into an already-confirmed rejection block would un-confirm it.
    Zone dup = na
    n0 = kind == "SR" or kind == "RJB" ? 0 : array.size(cx.zones)
    if n0 > 0
        for i = n0 - 1 to 0
            zx = array.get(cx.zones, i)
            if not zx.broken and zx.kind == kind and zx.bull == bull and zx.top >= zb and zx.bottom <= zt
                dup := zx
                break
    z = dup
    if na(dup)
        z := Zone.new(kind = kind, bull = bull, top = zt, bottom = zb, idx = idx, lastTouch = bar_index)
        if drawOn
            z.bx := box.new(f_xb(idx), z.top, bar_index, z.bottom, border_width = 1, extend = extend.right, text_size = v.zsz, text_halign = text.align_left, text_valign = text.align_top)
            f_zStyle(v, z)
        array.push(cx.zones, z)
    z

// §M11 — every directional reading a zone holds was measured for the side it had at the time.
// A flip, a breaker conversion or a propulsion re-arm invalidates all of them at once; leaving the
// inducement level in place is what made M37 (the highest-scored setup here) fire on evidence that
// belonged to the opposite direction.
f_zReset(z) =>
    z.idmLvl    := na
    z.idmSwept  := false
    z.nearMiss  := na
    z.nearSwept := false
    z.fiftyHit  := false

f_hiddenInterest(cf, cx, top, bottom, idx) =>
    clean = true
    n = array.size(cx.zones)
    if n > 0
        for i = 0 to n - 1
            z = array.get(cx.zones, i)
            if z.idx < idx and z.idx >= idx - cf.klLookLeft and z.top >= bottom and z.bottom <= top
                clean := false
                break
    clean

// ── L4 ZONE ENGINE — once per bar (R1); fed by L1 events + L2 strength; writes into Feat ──
f_zones(cf, cx, v, fv, aV, swH, swL, bullBOS, bearBOS, bullCHoCH, bearCHoCH, mvStrong, mvFvg, vOk, oIdxIn, onSR, onKL, onOB, onBrk, onRcb, onRjb, onQm) =>
    pIdx = bar_index - cf.swingLen
    // §6.1 S/R wick zones at every confirmed pivot (created always — KL scoring needs them; drawn only if onSR)
    if not na(swH)
        o = cf.swingLen
        f_zAdd(cf, cx, v, aV, "SR", false, high[o], math.max(open[o], close[o]), pIdx, onSR)
    if not na(swL)
        o = cf.swingLen
        f_zAdd(cf, cx, v, aV, "SR", true, math.min(open[o], close[o]), low[o], pIdx, onSR)

    // BOS → Key Level score on origin SR + Order Block / S&D
    if (bullBOS or bearBOS) and barstate.isconfirmed
        isBull = bullBOS
        // S2 — the leg origin from f_legOrigin (same one f_moveStrength measured), not the pivot
        // bookkeeping, which f_structure may already have shifted on this bar
        oIdx   = oIdxIn
        off = bar_index - oIdx
        if off >= 1 and off < 490
            n0 = array.size(cx.zones)
            if n0 > 0
                for i = 0 to n0 - 1
                    z = array.get(cx.zones, i)
                    // the origin low is not always a pivot bar: accept the origin-side SR zone within
                    // swingLen bars of it
                    if z.kind == "SR" and z.bull == isBull and z.kl == 0 and z.idx >= oIdx - cf.swingLen and z.idx <= oIdx + cf.swingLen
                        // v8.2.1 · base was 2 of 3, so "imbalance in the move" alone made KL and any
                        // second criterion made KL★. In a trend both are almost free (older zones sit
                        // at other prices, so `clean` is nearly always true) and the review charts
                        // carried ~10 KL★ each - which re-opened the "price is at a zone" gate that
                        // §srOk exists to close. Base 1 + the departure's own strength: KL now needs
                        // two earned criteria and KL★ needs three.
                        sc = 1
                        sc += mvFvg >= 1 ? 1 : 0
                        sc += mvStrong ? 1 : 0
                        sc += f_hiddenInterest(cf, cx, z.top, z.bottom, oIdx) ? 1 : 0
                        sc += cf.klUseVol and vOk ? 1 : 0
                        z.kl := sc
                        z.depStrong := mvStrong
                        if sc >= cf.klMinCrit and na(z.bx) and onKL
                            z.bx := box.new(f_xb(z.idx), z.top, bar_index, z.bottom, border_width = 1, extend = extend.right, text_size = v.zsz, text_halign = text.align_left, text_valign = text.align_top)
                        f_zStyle(v, z)
                        break
            // §C3 — this block used to be wrapped in `if onOB`, a "Show" toggle. Turning OB boxes
            // off therefore stopped the zones from existing at all and silently disabled M36, M37,
            // M42, M45, M48 and the whole breaker chain. Detection now always runs; `onOB` is passed
            // to f_zAdd as the draw flag, exactly like the SR zones below already did.
            if not mvStrong
                if isBull
                    fv.evtUp := "T5 no-FVG"
                    fv.tipUp := "Big move but NO imbalance — no OB here (Type 5). Look for reversal the other way."
                else
                    fv.evtDn := "T5 no-FVG"
                    fv.tipDn := "Big move but NO imbalance — no OB here (Type 5). Look for reversal the other way."
            else
                // S2 — OB = the last opposite-colour candle before the first displacement candle of
                // the leg (ICT definition). The old code took the pivot candle and the "G1" widening
                // was an always-true artefact of the T2 shift. Walk from the origin towards now.
                obOff = off
                obT   = "T1"
                dispOff = -1
                if obOff >= 2
                    for k = obOff - 1 to math.max(obOff - cf.swingLen, 1)
                        rk = high[k] - low[k]
                        bodyK = rk > 0 ? math.abs(close[k] - open[k]) / rk : 0.0
                        colOk = isBull ? close[k] > open[k] : close[k] < open[k]
                        if colOk and bodyK >= 0.5 and rk >= aV * 0.5
                            dispOff := k
                            break
                if dispOff > 0
                    // nearest opposite-colour candle between the displacement candle and the origin
                    for k = dispOff + 1 to obOff
                        opp = isBull ? close[k] < open[k] : close[k] > open[k]
                        if opp
                            obOff := k
                            obT   := k == off ? "T1" : "T1b"
                            break
                if high[obOff] <= high[obOff + 1] and low[obOff] >= low[obOff + 1]
                    obOff := obOff + 1
                    obT   := "T2"
                zTop = high[obOff]
                zBot = low[obOff]
                if obOff >= 1
                    nb = obOff - 1
                    smallBody = math.abs(close[obOff] - open[obOff]) < 0.3 * (high[obOff] - low[obOff])
                    engulfs   = low[nb] <= low[obOff] and high[nb] >= high[obOff]
                    if smallBody and engulfs
                        zTop := isBull ? high[obOff] : high[nb]
                        zBot := isBull ? low[nb] : low[obOff]
                        obT  := "G2"
                obIdx  = bar_index - obOff
                fvgLvl = f_fvgNear(cx, isBull, obIdx, obIdx + 3)
                if na(fvgLvl)
                    if isBull
                        fv.evtUp := "T5 no-FVG"
                        fv.tipUp := "STRONG move but no FVG aligned with the OB candle — OB invalid (Type 5)."
                    else
                        fv.evtDn := "T5 no-FVG"
                        fv.tipDn := "STRONG move but no FVG aligned with the OB candle — OB invalid (Type 5)."
                else
                    // the OB CANDLE's own extreme is what may have eaten into the gap
                    selfMit = isBull ? high[obOff] > fvgLvl : low[obOff] < fvgLvl
                    if selfMit
                        if isBull
                            fv.evtUp := "T3 invalid"
                            fv.tipUp := "OB wick mitigated its own FVG (Type 3) — not plotted."
                        else
                            fv.evtDn := "T3 invalid"
                            fv.tipDn := "OB wick mitigated its own FVG (Type 3) — not plotted."
                    else
                        sw = isBull ? cx.sweptLowBar : cx.sweptHighBar
                        hpNow = sw >= obIdx - cf.sweepWindow and sw <= obIdx + 2
                        // §M3 — obHpOnly used to be passed as the DRAW flag, so a non-HP order block
                        // was still created and still gated every OB setup: the input hid the box and
                        // changed nothing about the signals. Now it suppresses the zone itself.
                        if hpNow or not cf.obHpOnly
                            z = f_zAdd(cf, cx, v, aV, "OB", isBull, zTop, zBot, obIdx, onOB)
                            z.obType := obT
                            z.depStrong := mvStrong
                            z.hp := hpNow
                            fv.newObHp := hpNow
                            f_zStyle(v, z)

    // CHoCH → Reclaimed Block + QM pending
    // §C3 — `onRcb` / `onQm` are display toggles and used to gate detection: hiding the boxes
    // deleted M39 and M70 outright. They are now only draw flags.
    if bullCHoCH and barstate.isconfirmed
        // S4 — the reclaimed candle is the origin of the CHoCH leg (f_legOrigin), not a pivot guess
        llIdx = oIdxIn
        off = bar_index - llIdx
        if off >= 1 and off < 490 and llIdx > 0
            legMid = (low[off] + high) * 0.5
            if high[off] <= legMid or not cf.rcbFibGate
                zr = f_zAdd(cf, cx, v, aV, "RCB", true, high[off], low[off], llIdx, onRcb)
                zr.depStrong := cx.chochReal
        cx.qmBullPend := true
        // §6.10 the QM head is the MOST RECENT low; the prev-low form belongs to RCB (§6.8)
        cx.qmBullLL   := cx.curLow
        cx.qmBullAt   := bar_index
    if bearCHoCH and barstate.isconfirmed
        hhIdx = oIdxIn
        off = bar_index - hhIdx
        if off >= 1 and off < 490 and hhIdx > 0
            legMid = (high[off] + low) * 0.5
            if low[off] >= legMid or not cf.rcbFibGate
                zr = f_zAdd(cf, cx, v, aV, "RCB", false, high[off], low[off], hhIdx, onRcb)
                zr.depStrong := cx.chochReal
        cx.qmBearPend := true
        cx.qmBearHH   := cx.curHigh
        cx.qmBearAt   := bar_index
    if cx.qmBullPend
        if bar_index - cx.qmBullAt > cf.qmMaxBars or bearCHoCH
            cx.qmBullPend := false
        else if not na(cx.qmBullLL) and low < cx.qmBullLL and barstate.isconfirmed
            zq = f_zAdd(cf, cx, v, aV, "QM", true, cx.qmBullLL, low, bar_index, onQm)
            zq.depStrong := cx.chochReal
            cx.qmBullPend := false
    if cx.qmBearPend
        if bar_index - cx.qmBearAt > cf.qmMaxBars or bullCHoCH
            cx.qmBearPend := false
        else if not na(cx.qmBearHH) and high > cx.qmBearHH and barstate.isconfirmed
            zq = f_zAdd(cf, cx, v, aV, "QM", false, high, cx.qmBearHH, bar_index, onQm)
            zq.depStrong := cx.chochReal
            cx.qmBearPend := false

    // §6.9 Rejection Block — §C3: `onRjb` is a draw flag, not a detection gate (it used to kill M40)
    if barstate.isconfirmed and bar_index > 2
        rr1  = high[1] - low[1]
        uw1 = rr1 > 0 ? (high[1] - math.max(open[1], close[1])) / rr1 : 0.0
        dw1 = rr1 > 0 ? (math.min(open[1], close[1]) - low[1]) / rr1 : 0.0
        if uw1 >= cf.rjbWickMin
            zb = math.max(open[1], close[1])
            // R1 — a "weak" RJB (next candle re-entered the wick) was never tradable (R5) but still got a
            // box, IDM✓ and ★★ text on the review charts. Only create the block when the next candle held out.
            if high < zb
                z = f_zAdd(cf, cx, v, aV, "RJB", false, high[1], zb, bar_index - 1, onRjb)
                z.confirmed := false
                z.pendFvg   := 2
                f_zStyle(v, z)
        if dw1 >= cf.rjbWickMin
            zt = math.min(open[1], close[1])
            if low > zt
                z = f_zAdd(cf, cx, v, aV, "RJB", true, zt, low[1], bar_index - 1, onRjb)
                z.confirmed := false
                z.pendFvg   := 2
                f_zStyle(v, z)

    // per-bar maintenance + "zone at price" feature extraction (nearest active zone touched this bar)
    n = array.size(cx.zones)
    bestPri  = -1
    bestPriS = -1
    if n > 0
        for i = n - 1 to 0
            z = array.get(cx.zones, i)
            changed = false
            if z.broken
                if bar_index - z.brokenAt > 20
                    if not na(z.bx)
                        box.delete(z.bx)
                    array.remove(cx.zones, i)
                continue
            if z.idx >= bar_index
                continue

f_htf(cf, cx, sx, fv, isNew, o1, h1, l1, c1, o2, h2, l2, c2, ph, pl, h3, l3) =>
    if isNew and not na(c1)
        cx.hBarCount := cx.hBarCount + 1
        // P1-B — HTF FVG on the three completed candles: bull = low[1] above high[3], bear = mirror
        if not na(h3) and l1 > h3
            sx.hFvgBT := l1
            sx.hFvgBB := h3
            sx.hFvgBBar := cx.hBarCount
        if not na(l3) and h1 < l3
            sx.hFvgST := l3
            sx.hFvgSB := h1
            sx.hFvgSBar := cx.hBarCount
        // v8.5.0 W16 — an HTF gap that is months old is not "the higher-timeframe reason for the trade"
        if not na(sx.hFvgBB) and cx.hBarCount - sx.hFvgBBar > 50
            sx.hFvgBT := na
            sx.hFvgBB := na
        if not na(sx.hFvgST) and cx.hBarCount - sx.hFvgSBar > 50
            sx.hFvgST := na
            sx.hFvgSB := na
        if not na(cx.hCurHigh) and not cx.hHiBroken and h1 > cx.hCurHigh and c1 <= cx.hCurHigh
            cx.sweepUpAt := cx.hBarCount
        if not na(cx.hCurLow) and not cx.hLoBroken and l1 < cx.hCurLow and c1 >= cx.hCurLow
            cx.sweepDnAt := cx.hBarCount
        // S4 — same rules as the LTF engine: a break only moves the bias when the HTF candle shows
        // displacement (body ≥ half its range); a counter-bias pivot break is internal; the bias
        // flips only on a close through the protected swing (last pivot before the breaking leg).
        dispH = (h1 - l1) > 0 and math.abs(c1 - o1) / (h1 - l1) >= 0.5
        bBrk = not na(cx.hCurHigh) and not cx.hHiBroken and c1 > cx.hCurHigh
        sBrk = not na(cx.hCurLow)  and not cx.hLoBroken and c1 < cx.hCurLow
        // v8.5.0 D4 — a break without displacement used to consume the pivot here, so the displaced
        // follow-through on the next HTF candle found nothing to break. Consume only on a displaced break.
        if bBrk and dispH
            cx.hHiBroken := true
        if sBrk and dispH
            cx.hLoBroken := true
        // R1 — the first BULL BOS of a chart can happen before any HTF pivot low exists, which left
        // hMajorLow = na and the bias stuck at BULL through a two-week decline (gold review chart).
        // Fall back to the last pivot as the protected swing.
        protH = nz(cx.hMajorHigh, cx.hCurHigh)
        protL = nz(cx.hMajorLow,  cx.hCurLow)
        // R2 — LH + close through the last pivot low (or HL + close through the last pivot high) is a
        // change of character even when the protected swing sits far away
        lhH = not na(cx.hPrevHigh) and not na(cx.hCurHigh) and cx.hCurHigh < cx.hPrevHigh
        hlH = not na(cx.hPrevLow)  and not na(cx.hCurLow)  and cx.hCurLow  > cx.hPrevLow
        chUpH = cx.hTrend == "BEAR" and ((not na(protH) and c1 > protH) or (bBrk and hlH))
        chDnH = cx.hTrend == "BULL" and ((not na(protL) and c1 < protL) or (sBrk and lhH))
        if dispH
            if chUpH
                cx.hTrend    := "BULL"
                cx.hMajorLow := cx.hCurLow
            else if chDnH
                cx.hTrend     := "BEAR"
                cx.hMajorHigh := cx.hCurHigh
            else if bBrk and cx.hTrend != "BEAR"
                cx.hTrend    := "BULL"
                cx.hMajorLow := cx.hCurLow
            else if sBrk and cx.hTrend != "BULL"
                cx.hTrend     := "BEAR"
                cx.hMajorHigh := cx.hCurHigh
        if not na(ph)
            cx.hPrevHigh := cx.hCurHigh
            cx.hCurHigh  := ph
            cx.hHiBroken := false
            if cx.hTrend == "BEAR" and na(cx.hMajorHigh)
                cx.hMajorHigh := ph
        if not na(pl)
            cx.hPrevLow  := cx.hCurLow
            cx.hCurLow   := pl
            cx.hLoBroken := false
            if cx.hTrend == "BULL" and na(cx.hMajorLow)
                cx.hMajorLow := pl
    // P1-B — a close through the far edge mitigates the HTF FVG; otherwise flag price inside it
    if not na(sx.hFvgBB) and close < sx.hFvgBB
        sx.hFvgBT := na
        sx.hFvgBB := na
    if not na(sx.hFvgST) and close > sx.hFvgST
        sx.hFvgST := na
        sx.hFvgSB := na
    fv.inHtfFvgB := not na(sx.hFvgBB) and low <= sx.hFvgBT and high >= sx.hFvgBB
    fv.inHtfFvgS := not na(sx.hFvgST) and high >= sx.hFvgSB and low <= sx.hFvgST
    fv.htfBias    := cx.hTrend
    fv.htfSweepUp := cx.hBarCount - cx.sweepUpAt < cf.sweepValidHTF
    fv.htfSweepDn := cx.hBarCount - cx.sweepDnAt < cf.sweepValidHTF
    if not na(c1) and not na(c2)
        rr1  = h1 - l1
        upW = rr1 > 0 ? (h1 - math.max(o1, c1)) / rr1 : 0.0
        dnW = rr1 > 0 ? (math.min(o1, c1) - l1) / rr1 : 0.0
        fv.htfManipBull := l1 < l2 and c1 > h2 and c1 > o1 and dnW >= cf.manipWickRatio
        fv.htfManipBear := h1 > h2 and c1 < l2 and c1 < o1 and upW >= cf.manipWickRatio
        fv.htfRejBear   := upW >= cf.rejWickRatio and c1 < o1
        fv.htfRejBull   := dnW >= cf.rejWickRatio and c1 > o1

// nearest opposite reaction zone / FVG beyond price (TP1) — event-only scans
// nearest opposite level that is AT LEAST minDist beyond px. Taking the nearest one of ANY distance
// is what made TP1 a ~1R target: with 30 zones and 40 FVGs tracked there is almost always one close.
// S8 (audit B8) — target QUALITY: an unscored pivot wick zone or an unconfirmed / weak rejection
// block is not a target (with 30-60 zones retained one was almost always in range, which defeated
// the "real target" rule). Qualified liquidity (pool / PD / PW / SES) is a valid TP1 too.
f_nearestOpp(cf, cx, isLong, px, minDist, maxDist) =>
    best = float(na)
    nzc = array.size(cx.zones)
    if nzc > 0
        for i = 0 to nzc - 1
            z = array.get(cx.zones, i)
            if z.broken or z.bull == isLong
                continue
            qualZ = z.kind == "SR" ? z.kl >= cf.klMinCrit : (z.confirmed and not z.weak)
            if not qualZ
                continue
            lvl = isLong ? z.bottom : z.top
            dz = isLong ? lvl - px : px - lvl
            closer = isLong ? lvl < best : lvl > best
            if dz >= minDist and dz <= maxDist and (na(best) or closer)
                best := lvl
    nf = array.size(cx.fvgs)
    if nf > 0
        for i = 0 to nf - 1
            g = array.get(cx.fvgs, i)
            if g.mitigated or g.isBull == isLong
                continue
            lvl = isLong ? g.bottom : g.top
            dg = isLong ? lvl - px : px - lvl
            closer = isLong ? lvl < best : lvl > best
            if dg >= minDist and dg <= maxDist and (na(best) or closer)
                best := lvl
    nl = array.size(cx.lv)
    if nl > 0
        for i = 0 to nl - 1
            L = array.get(cx.lv, i)
            if L.done or L.isHigh != isLong
                continue
            qualL = L.pooled or L.kind == "PDH" or L.kind == "PDL" or L.kind == "PWH" or L.kind == "PWL" or L.kind == "SESH" or L.kind == "SESL" or (cf.sweepExtOk and L.ext)
            if not qualL
                continue
            dl = isLong ? L.price - px : px - L.price
            closer = isLong ? L.price < best : L.price > best
            if dl >= minDist and dl <= maxDist and (na(best) or closer)
                best := L.price
    best

// P1-4 (audit F4) — is the road to TP1 clear? f_nearestOpp only looks inside the [minRR, maxRR]
// band, so a qualified opposing zone or FVG sitting CLOSER than minRR is invisible to it: the
// target passes the R:R filter while price must trade through a block to reach it.
// v8.5.0 W5 — returns 0 clear · 2 = a qualified opposing zone or unmitigated FVG in the path (STRICT
// rejects) · 1 = qualified liquidity in the path (WARN only: a pool may simply be run through).
f_pathBlock(cf, cx, isLong, px, tp, aV) =>
    tol = aV * 0.10
    hit = 0
    nzc = array.size(cx.zones)
    if nzc > 0
        for i = 0 to nzc - 1
            z = array.get(cx.zones, i)
            if z.broken or z.bull == isLong
                continue
            qualZ = z.kind == "SR" ? z.kl >= cf.klMinCrit : (z.confirmed and not z.weak)
            lvl = isLong ? z.bottom : z.top
            if qualZ and ((isLong and lvl > px + tol and lvl < tp - tol) or (not isLong and lvl < px - tol and lvl > tp + tol))
                hit := 2
                break
    nf = array.size(cx.fvgs)
    if hit == 0 and nf > 0
        for i = 0 to nf - 1
            g = array.get(cx.fvgs, i)
            if g.mitigated or g.isBull == isLong
                continue
            lvl = isLong ? g.bottom : g.top
            if (isLong and lvl > px + tol and lvl < tp - tol) or (not isLong and lvl < px - tol and lvl > tp + tol)
                hit := 2
                break
    nl = array.size(cx.lv)
    if hit == 0 and nl > 0
        for i = 0 to nl - 1
            L = array.get(cx.lv, i)
            if L.done or L.isHigh != isLong
                continue
            qualL = L.pooled or L.kind == "PDH" or L.kind == "PDL" or L.kind == "PWH" or L.kind == "PWL" or L.kind == "SESH" or L.kind == "SESL"
            if qualL and ((isLong and L.price > px + tol and L.price < tp - tol) or (not isLong and L.price < px - tol and L.price > tp + tol))
                hit := 1
                break
    hit

// nearest unresolved liquidity beyond a price (TP2: EQH/EQL · BSL/SSL · PDH/PDL · PWH/PWL)
f_nearestLiq(cx, isLong, px) =>
    best = float(na)
    n = array.size(cx.lv)
    if n > 0
        for i = 0 to n - 1
            L = array.get(cx.lv, i)
            if L.done or L.isHigh != isLong
                continue
            if (isLong and L.price > px and (na(best) or L.price < best)) or (not isLong and L.price < px and (na(best) or L.price > best))
                best := L.price
    best

// §9.4 confluence score (spec is the authority; setup base score only chose the candidate)
// takes the trend as a string (§M9: the caller decides whether that is this bar's trend or the
// trend as the bar opened) instead of the whole Struct, which it only ever used for `.trend`
f_conf(fv, trendStr, isLong) =>
    zz = isLong ? fv.zb : fv.zs          // S9 — the trade's own side
    sc = 1
    sc += (isLong and fv.htfBias == "BULL") or (not isLong and fv.htfBias == "BEAR") ? 1 : 0
    sc += (isLong and trendStr == "BULLISH") or (not isLong and trendStr == "BEARISH") ? 1 : 0
    // R2 — fading the LTF trend costs a point (it used to be merely "no bonus")
    // v8.5.2 E-2 — not when the setup IS the HTF-aligned pullback: HTF aligned + discount (premium) + POI tapped
    pbWv = i_ltfPb and zz.kind != "" and ((isLong and fv.htfBias == "BULL" and fv.inDiscount) or (not isLong and fv.htfBias == "BEAR" and fv.inPremium))
    sc -= not pbWv and ((isLong and trendStr == "BEARISH") or (not isLong and trendStr == "BULLISH")) ? 1 : 0
    // P1-3 (audit F6) — the four POI terms below all describe the SAME zone: one fresh, well-scored
    // block that departed strongly and overlaps an FVG used to earn +4 for a single piece of
    // evidence. The POI category is capped at +2. Liquidity targets and the sweep / IDM term are
    // independent evidence and stay uncapped.
    poiPts = (zz.depStrong and zz.arr == 1 ? 1 : 0) + (zz.kind != "" and zz.touches <= 1 ? 1 : 0) +
             (zz.kl >= 4 ? 1 : 0) +
             (zz.kind != "" and ((isLong and fv.atBullFvg) or (not isLong and fv.atBearFvg)) ? 1 : 0)
    sc += math.min(poiPts, 2)
    // v8.5.0 W9 — one unresolved level within 3 ATR is nearly always there with 30 levels tracked; two is evidence
    sc += (isLong and fv.bslAbove >= 2) or (not isLong and fv.sslBelow >= 2) ? 1 : 0
    sc += (isLong and (fv.sweptLowQ or zz.idm)) or (not isLong and (fv.sweptHighQ or zz.idm)) ? 1 : 0
    sc -= zz.arr == -1 ? 2 : 0               // v8.5.1 S1 — the trade side's own arrival
    sc -= zz.sweepReq and not ((isLong and fv.sweptLow) or (not isLong and fv.sweptHigh)) ? 2 : 0
    sc -= trendStr == "RANGE" ? 1 : 0
    // P1-B — the LTF POI sits inside an unmitigated HTF FVG: the higher-timeframe reason for the trade
    sc += (isLong and fv.inHtfFvgB) or (not isLong and fv.inHtfFvgS) ? 1 : 0
    sc += not na(fv.htfPdPct) and ((isLong and fv.htfPdPct <= 0.5) or (not isLong and fv.htfPdPct >= 0.5)) ? 1 : 0   // v8.5.2 E-7 — HTF discount / premium
    // v8.5.0 W9 — ONE location term. OTE (62-79 % retracement) sits inside discount, and the side of the
    // NY midnight open is correlated with both, so the three used to pay +3 for one fact.
    // OTE +2 · discount / premium +1 · deep premium / discount −1 · NY-open side only as a tie-break.
    locPts = 0
    if i_pdOn
        ote = isLong ? (fv.pdPct >= 0.21 and fv.pdPct <= 0.38) : (fv.pdPct >= 0.62 and fv.pdPct <= 0.79)
        locPts := ote and not fv.pdBeyond ? 2 :
                  (isLong and fv.inDiscount) or (not isLong and fv.inPremium) ? 1 :
                  (isLong and fv.pdPct > i_pdDeep) or (not isLong and fv.pdPct < 1.0 - i_pdDeep) ? -1 : 0
    if i_nyOpen and not na(nyOpen) and locPts == 0
        locPts := (isLong and close < nyOpen) or (not isLong and close > nyOpen) ? 1 : 0
    sc += locPts
    // NEW · killzone — the London and NY windows are where the sweeps this engine hunts get run.
    // v8.5.0 W9 — while the killzone FILTER is on every signal is already inside one, so the point is noise.
    if i_ssnScore and not ssnFilt
        sc += fv.sessName == "London" or fv.sessName == "NY" ? 1 : 0
    sc

// §9.2 SL: 1 zone-based → 2 sweep-based → 3 major HL/LH (if within maxSLATR) → fallback candle extreme
// P0-2 (audit F3) — the stop takes the FINAL ENTRY price (`ep`), not `close`. It used to know only
// the close, so a RETEST / BOTH entry moved to the swept level measured its invalidation, its
// distance cap and its risk from a price the trade never used.
// v8.4.1 H3 — `x` is the setup extreme (this candle's low / high in IMMEDIATE mode, the lowest low / highest
// high from the POI tap to the confirming bar otherwise). The stop sits beyond the zone AND beyond x: a
// reaction wick through the zone used to leave the stop inside a low the market had already printed.
// v8.5.0 D2 — when this bar carries a sweep, the true wick extreme (pend path included) joins `x`.
f_stop(cf, cx, fv, isLong, aV, bufA, maxA, ep, x) =>
    buf = aV * bufA
    xx = isLong ? math.min(x, nz(fv.sweptExtLo, x)) : math.max(x, nz(fv.sweptExtHi, x))
    sl = float(na)
    // S9 — the trade's own-side zone defines the stop (the slot is per side now)
    zz = isLong ? fv.zb : fv.zs
    if zz.kind != ""
        sl := isLong ? math.min(zz.bottom, xx) - buf : math.max(zz.top, xx) + buf
    else if (isLong and fv.sweptLow) or (not isLong and fv.sweptHigh)
        sl := isLong ? xx - buf : xx + buf
    else
        maj = isLong ? cx.majorLow : cx.majorHigh
        if not na(maj) and math.abs(ep - maj) <= aV * maxA
            sl := isLong ? math.min(maj, xx) - buf : math.max(maj, xx) + buf
        else
            sl := isLong ? xx - buf : xx + buf
    // sanity: SL must sit on the correct side of the ENTRY (a zone edge can be beyond it when the
    // touched zone is the opposite kind, or too wide vs maxA) -> fall back to the candle extreme
    if (isLong and (sl >= ep or ep - sl > aV * maxA * 2)) or (not isLong and (sl <= ep or sl - ep > aV * maxA * 2))
        sl := isLong ? math.min(xx, ep) - buf : math.max(xx, ep) + buf
    sl

// build one Sig (entry / SL / TP1 / TP2 / score / grade)
f_sig(cf, cx, fv, trendStr, name, tier, note, isLong, aV, eMode, slBufA, maxSLA, fixedR, fixedRm, minRR, maxRR, rrReal, pathMode, ePx, minRiskT, minRiskA, x, lim) =>
    g = Sig.new()
    g.name  := name
    g.dir   := isLong ? 1 : -1
    g.entry := close
    // P1-A — limit entry back at the POI instead of the confirming candle's close
    zzE = isLong ? fv.zb : fv.zs
    if ePx == "ZONE EDGE" and zzE.kind != ""
        g.entry := isLong ? zzE.top : zzE.bottom
        g.note  := "LIMIT @ zone edge · "
    else if ePx == "CANDLE 50%" or ePx == "ZONE EDGE"
        g.entry := (high + low) * 0.5
        g.note  := "LIMIT @ candle 50% · "
    m10px = isLong ? fv.sweptPriceLo : fv.sweptPriceHi
    if name == "M10" and eMode != "IMMEDIATE" and not na(m10px)
        g.entry := m10px
        g.note  := eMode == "RETEST" ? "RETEST limit @ swept level · " : "BOTH: immediate + retest limit @ level · "
    // v8.5.2 E-4 — M74: the caller hands in the limit (leg FVG CE / leg 50 %); SL, risk, TP, path all follow it
    if not na(lim)
        g.entry := lim
        g.note  := "MSS → FVG limit · "
    // P0-2 (audit F3) — fixed order: FINAL ENTRY -> invalidation -> risk -> targets -> R:R.
    // f_stop now receives the entry, so the wrong-side re-check that used to patch up the RETEST
    // entry afterwards is gone: the stop is built against the price the trade actually uses.
    g.sl := f_stop(cf, cx, fv, isLong, aV, slBufA, maxSLA, g.entry, x)
    risk = math.abs(g.entry - g.sl)
    // P2-B — floor the risk at minRiskT ticks (and at 0.5 ATR when the stop collapsed onto the entry)
    minR = math.max(syminfo.mintick * minRiskT, aV * minRiskA, risk <= 0 ? aV * 0.5 : 0.0)   // v8.4.1 M1
    if risk < minR
        risk := minR
        g.sl := isLong ? g.entry - risk : g.entry + risk
    okRR = true
    if fixedR
        g.tp1 := isLong ? g.entry + fixedRm * risk : g.entry - fixedRm * risk
        g.tp2 := isLong ? g.tp1 + risk : g.tp1 - risk
        g.tp3 := isLong ? g.tp2 + risk : g.tp2 - risk
        g.tgt := "FIXED-R"                      // P0-4 — a risk multiple, never a market level
    else
        // TP1 = nearest real opposite level at least minRR away. The old rule took the nearest level
        // of ANY distance and only demanded 1R, so the average winner was ~1R while every loser was
        // 1R — negative expectancy below a 50% win rate no matter how good the setup was.
        need = risk * minRR
        cap  = risk * maxRR
        // v8.5.0 W4 — the scan has no ceiling any more: a real level beyond Max R:R is still the target,
        // marked "FAR" (a tight-stop entry at the POI edge was being REJECTED for having its target too far)
        tpA  = f_nearestOpp(cf, cx, isLong, g.entry, need, 1.0e10)
        // §C4 — the confluence score, the HTF/trend gate and this R:R test all depend on the
        // direction and the feature vector, never on WHICH setup won the chain, so they cannot
        // single out one candidate: if they reject the top pick they would reject every other
        // candidate on that side too, and no second-best slot would help. The single real
        // exception is M10 in RETEST / BOTH mode, which moves the entry to the swept level and so
        // measures a different risk. Fall back to the immediate entry instead of dropping it.
        // v8.5.0 D10 — M10 only: a limit-entry mode (P1-A) with no target is rejected, not silently
        // turned into a market entry.
        if na(tpA) and g.entry != close and name == "M10"
            g.entry := close
            g.note  := g.note + "R:R fallback → IMMEDIATE entry · "
            g.sl    := f_stop(cf, cx, fv, isLong, aV, slBufA, maxSLA, g.entry, x)
            risk    := math.abs(g.entry - g.sl)
            if risk <= 0
                risk := aV * 0.5
                g.sl := isLong ? g.entry - risk : g.entry + risk
            need := risk * minRR
            cap  := risk * maxRR
            tpA  := f_nearestOpp(cf, cx, isLong, g.entry, need, 1.0e10)
        if na(tpA)
            if rrReal
                okRR := false                       // no real target that far -> not worth taking
            else
                // P0-4 (audit F4) — this target is a risk multiple pointing at whatever happens to
                // be there, which is usually empty space. It is now MARKED instead of being shown
                // exactly like a structural one.
                tpA   := isLong ? g.entry + need : g.entry - need
                g.tgt := "SYNTH"
        else
            g.tgt := math.abs(tpA - g.entry) > cap ? "FAR" : "REAL"
        g.tp1 := tpA
        // each further target must clear the previous one by at least 1R, so the ladder advances
        tpB = f_nearestLiq(cx, isLong, na(g.tp1) ? g.entry : g.tp1)
        g.tp2 := not na(tpB) and ((isLong and tpB >= g.tp1 + risk) or (not isLong and tpB <= g.tp1 - risk)) ? tpB : (isLong ? g.tp1 + risk : g.tp1 - risk)
        tpC = f_nearestLiq(cx, isLong, na(g.tp2) ? g.entry : g.tp2)
        g.tp3 := not na(tpC) and ((isLong and tpC >= g.tp2 + risk) or (not isLong and tpC <= g.tp2 - risk)) ? tpC : (isLong ? g.tp2 + risk : g.tp2 - risk)
    g.rr    := risk > 0 and not na(g.tp1) ? math.abs(g.tp1 - g.entry) / risk : 0.0
    // P1-4 (audit F4) — obstacle check on the FINAL entry -> TP1 path
    blkRaw = f_pathBlock(cf, cx, isLong, g.entry, g.tp1, aV)
    blk = pathMode != "OFF" and not na(g.tp1) ? blkRaw : 0
    pathPen = 0
    if blk > 0
        // v8.5.0 W5 — 2 = zone / FVG (STRICT rejects) · 1 = liquidity pool (always WARN)
        g.note := g.note + (blk == 2 ? "⚠ opposing zone/FVG between entry and TP1 · " : "⚠ liquidity pool between entry and TP1 · ")
        if pathMode == "STRICT" and blk == 2
            okRR := false
        else
            pathPen := 1
    // P0-3 (audit F3 / F4) — hard geometry gate on the FINAL numbers. A stop on the wrong side of
    // the entry, a target on the wrong side, or an R:R that no longer clears the minimum after the
    // entry moved is not a tradable description of the setup, so the signal is dropped, not shown.
    if na(g.sl) or na(g.tp1) or (isLong and (g.sl >= g.entry or g.tp1 <= g.entry)) or (not isLong and (g.sl <= g.entry or g.tp1 >= g.entry)) or (not fixedR and g.rr < minRR - 0.01)
        okRR := false
    g.score := tier * 2 + f_conf(fv, trendStr, isLong) - pathPen      // S6 · P1-4
    g.note  := g.note + note + (g.tgt == "REAL" ? "" : " · ⚠ TP1 = " + g.tgt)
    // an empty name hides the signal everywhere: label, trade lines, alerts and the panel
    g.name  := okRR ? name : ""
    g

// P0-5 (audit F10) — setup identity: the module, the side, and the POI it is built on (the tapped
// zone, or the swept level when the setup has no zone). Two signals with the same identity are the
// same trade seen twice, however many bars apart they print.
f_ident(g, fv) =>
    zz  = g.dir == 1 ? fv.zb : fv.zs
    anc = zz.kind != "" ? (zz.top + zz.bottom) * 0.5 : nz(g.dir == 1 ? fv.sweptPriceLo : fv.sweptPriceHi, 0.0)
    g.name + (g.dir == 1 ? "L" : "S") + zz.kind + str.tostring(anc, format.mintick)
