// ═════════════════════════════════════════════════════════════════ STEP 2
// CORE SERIES + CANDLE PSYCHOLOGY (wick / body anatomy, confirmations)
// ═════════════════════════════════════════════════════════════════════════
atr     = ta.atr(atrLen)
body    = math.abs(close - open)
rng     = high - low
upWick  = high - math.max(open, close)
dnWick  = math.min(open, close) - low
bullBar = close > open
bearBar = close < open

// Signals & state changes only commit on the bar's final tick (or historical bars)
confirmed = barstate.isconfirmed

emaTrend  = ta.ema(close, trendLen)
chartBias = close > emaTrend ? 1 : -1

// Confirmation candles (Pin Bar / Engulfing / Strong Rejection)
bullPin    = rng > 0 and dnWick >= 2 * body and upWick <= body and dnWick >= minWickAtr * atr
bearPin    = rng > 0 and upWick >= 2 * body and dnWick <= body and upWick >= minWickAtr * atr
bullEngulf = bullBar and bearBar[1] and close >= open[1] and open <= close[1] and body > body[1]
bearEngulf = bearBar and bullBar[1] and close <= open[1] and open >= close[1] and body > body[1]
bullConfirm = bullPin or bullEngulf
bearConfirm = bearPin or bearEngulf
doji        = rng > 0 and body <= dojiPct / 100 * rng

// ---- CISD: change in state of delivery (ADD-4) --------------------------------
// Track the OPEN of the first candle of the most recent consecutive bull run
// and bear run. A confirmed close through that origin open = delivery flipped.
// Fires once per run (latched), resets when a fresh opposite run starts.
// FIX-16: strict consecutive delivery — a neutral candle (≤ CISD_NEUTRAL_PCT
// body) is neither an up- nor a down-close and ENDS both series; the next
// CISD needs a fresh run. Series are counted on the neutral-filtered flags.
cisdNeutral = rng == 0 or body <= CISD_NEUTRAL_PCT * rng
cisdBull    = bullBar and not cisdNeutral
cisdBear    = bearBar and not cisdNeutral
var float bullRunOpen = na
var float bearRunOpen = na
var int   bullRunLen  = 0
var int   bearRunLen  = 0
var bool  cisdDnDone  = false
var bool  cisdUpDone  = false
if confirmed
    if cisdNeutral                         // neutral candle → both series void
        bullRunOpen := na
        bearRunOpen := na
        bullRunLen  := 0
        bearRunLen  := 0
    if cisdBull and not cisdBull[1]        // new bull series begins here
        bullRunOpen := open
        bullRunLen  := 1
        cisdDnDone  := false
    else if cisdBull
        bullRunLen += 1
    if cisdBear and not cisdBear[1]        // new bear series begins here
        bearRunOpen := open
        bearRunLen  := 1
        cisdUpDone  := false
    else if cisdBear
        bearRunLen += 1
// Valid CISD needs a real series (≥2 candles) and a DISPLACEMENT-grade close
// through its origin open: body ≥ BRK_BODY_ATR AND the close in the outer
// STRONG_CLOSE_PCT of its own range (FIX-16). Noise flips in chop don't count.
cisdDn = confirmed and not na(bullRunOpen) and not cisdDnDone and cisdBear and close < bullRunOpen and bullRunLen >= 2 and body >= BRK_BODY_ATR * atr and close <= low + STRONG_CLOSE_PCT * rng
cisdUp = confirmed and not na(bearRunOpen) and not cisdUpDone and cisdBull and close > bearRunOpen and bearRunLen >= 2 and body >= BRK_BODY_ATR * atr and close >= high - STRONG_CLOSE_PCT * rng
if cisdDn
    cisdDnDone := true
if cisdUp
    cisdUpDone := true
// chip markers — dark pill background so text survives any candle color
if showCisd and cisdDn
    f_chip(false, "CISD", color.new(COL_BEAR, 55), color.white, mkSize)
if showCisd and cisdUp
    f_chip(true, "CISD", color.new(COL_BULL, 55), color.white, mkSize)

// ---- IFC candle (ADD-8): institutional funding candle -------------------------
// Grabs the prior candle's extreme (liquidity), then closes past the prior
// candle's MIDPOINT with a displacement body — smart money funding a move.
// FIX-17: the grab must take a REAL pool — the prior IFC_LOOK-bar extreme, not
// just the previous candle's wick — with meaningful penetration, and the close
// must land in the outer STRONG_CLOSE_PCT of the candle (not merely past the
// prior midpoint). Structural intent is supplied by the engines that consume
// the flag (zone confirmation, FEM), not asserted here.
prevMid  = (high[1] + low[1]) / 2
ifcPoolL = ta.lowest(low, IFC_LOOK)[1]
ifcPoolH = ta.highest(high, IFC_LOOK)[1]
ifcBull = low < ifcPoolL and ifcPoolL - low >= NEAR_LVL_ATR * atr and bullBar and close > prevMid and close >= high - STRONG_CLOSE_PCT * rng and body >= BRK_BODY_ATR * atr
ifcBear = high > ifcPoolH and high - ifcPoolH >= NEAR_LVL_ATR * atr and bearBar and close < prevMid and close <= low + STRONG_CLOSE_PCT * rng and body >= BRK_BODY_ATR * atr
if showIFC and ifcBull and confirmed
    f_chip(true, "IFC", color.new(color.gray, 55), color.new(COL_BULL, 0), mkSize)
if showIFC and ifcBear and confirmed
    f_chip(false, "IFC", color.new(color.gray, 55), color.new(COL_BEAR, 0), mkSize)

// Confirmation used by the zone / trendline engines: candle pattern, plus
// CISD when the user opts in — CISD is itself a displacement-grade confirm.
// An IFC candle in the right direction also qualifies (it IS the grab+go).
zoneBullConf = bullConfirm or ifcBull or (cisdAsConf and cisdUp)
zoneBearConf = bearConfirm or ifcBear or (cisdAsConf and cisdDn)

// ═════════════════════════════════════════════════════════════════ STEP 3
// KEY LEVELS (swing pivots) + LIQUIDITY SWEEPS
// ═════════════════════════════════════════════════════════════════════════
ph = ta.pivothigh(high, pivLen, pivLen)
pl = ta.pivotlow(low, pivLen, pivLen)

var float lastRes    = na   // last confirmed swing high  → buy-side liquidity above
var float lastSup    = na   // last confirmed swing low   → sell-side liquidity below
var int   lastResBar = na   // BAR of that swing high (dealing-range direction, FIX-4)
var int   lastSupBar = na   // BAR of that swing low
var float prevPH     = na   // previous pivot high (equal-highs detection)
var float prevPL     = na   // previous pivot low  (equal-lows detection)
// FIX-2: once-per-level break latches. A level can only "break" once; the
// latch resets when a NEW pivot replaces the level. This kills whipsaw BOS
// spam AND registers breaks that complete during pivot-confirmation lag.
var bool  resBroken  = false
var bool  supBroken  = false

// ⑪ dynamic trendline state — anchor pair, slope, touch count, line object
var int   prevPHBar = na
var int   prevPLBar = na
var float tlUpSlope = na   // ascending support line (higher lows)
var float tlUpPrice = na
var int   tlUpBar   = na
var int   tlUpTouch = 0
var line  tlUpLn    = na
var float tlDnSlope = na   // descending resistance line (lower highs)
var float tlDnPrice = na
var int   tlDnBar   = na
var int   tlDnTouch = 0
var line  tlDnLn    = na

// order-flow zigzag state (ADD-10): connect successive internal pivots
var float zzPrevP = na
var int   zzPrevB = na
var zzLines = array.new<line>()

// Equal highs / lows = engineered liquidity pools (resting stop clusters)
var eqhLvls = array.new<float>()
var eqhLns  = array.new<line>()
var eqlLvls = array.new<float>()
var eqlLns  = array.new<line>()

if not na(ph) and confirmed
    if showEq and not na(prevPH) and math.abs(ph - prevPH) <= eqTol * atr
        lvl = math.max(ph, prevPH)
        array.push(eqhLvls, lvl)
        array.push(eqhLns, line.new(bar_index - pivLen, lvl, bar_index + 1, lvl,
             color = color.new(COL_BEAR, 25), style = line.style_dotted, width = 2))
        if array.size(eqhLvls) > MAX_EQ_POOLS
            line.delete(array.get(eqhLns, 0))
            array.remove(eqhLns, 0)
            array.remove(eqhLvls, 0)
    // ⑪ lower high → (re)anchor the descending trendline off the last two
    // pivot highs; a confirmed close above the line kills it elsewhere
    // FIX-19: a VALIDATED line (≥1 touch) survives new lower highs — the new
    // pivot is another touch on it or a shallower pullback under it. Only a
    // candidate (untouched) or dead line re-anchors to the newest pivot pair.
    if not na(prevPH) and not na(prevPHBar) and ph < prevPH and (na(tlDnSlope) or tlDnTouch == 0)
        line.delete(tlDnLn)
        tlDnLn    := na
        tlDnSlope := (ph - prevPH) / (bar_index - pivLen - prevPHBar)
        tlDnPrice := ph
        tlDnBar   := bar_index - pivLen
        tlDnTouch := 0
        if showTL
            tlDnLn := line.new(prevPHBar, prevPH, bar_index - pivLen, ph,
                 color = color.new(COL_BEAR, 30), width = 2, extend = extend.right,
                 style = tlValidate ? line.style_dotted : line.style_solid)
    // order-flow zigzag: leg into this swing high
    if showOF
        if not na(zzPrevP)
            array.push(zzLines, line.new(zzPrevB, zzPrevP, bar_index - pivLen, ph,
                 color = ph > zzPrevP ? color.new(COL_BULL, 35) : color.new(COL_BEAR, 35), width = 1))
            if array.size(zzLines) > 40
                line.delete(array.get(zzLines, 0))
                array.remove(zzLines, 0)
        zzPrevP := ph
        zzPrevB := bar_index - pivLen
    prevPH     := ph
    prevPHBar  := bar_index - pivLen
    lastRes    := ph
    lastResBar := bar_index - pivLen
    resBroken  := false
if not na(pl) and confirmed
    if showEq and not na(prevPL) and math.abs(pl - prevPL) <= eqTol * atr
        lvl = math.min(pl, prevPL)
        array.push(eqlLvls, lvl)
        array.push(eqlLns, line.new(bar_index - pivLen, lvl, bar_index + 1, lvl,
             color = color.new(COL_BULL, 25), style = line.style_dotted, width = 2))
        if array.size(eqlLvls) > MAX_EQ_POOLS
            line.delete(array.get(eqlLns, 0))
            array.remove(eqlLns, 0)
            array.remove(eqlLvls, 0)
    // ⑪ higher low → (re)anchor the ascending trendline off the last two
    // pivot lows; a confirmed close below the line kills it elsewhere
    if not na(prevPL) and not na(prevPLBar) and pl > prevPL and (na(tlUpSlope) or tlUpTouch == 0)
        line.delete(tlUpLn)
        tlUpLn    := na
        tlUpSlope := (pl - prevPL) / (bar_index - pivLen - prevPLBar)
        tlUpPrice := pl
        tlUpBar   := bar_index - pivLen
        tlUpTouch := 0
        if showTL
            tlUpLn := line.new(prevPLBar, prevPL, bar_index - pivLen, pl,
                 color = color.new(COL_BULL, 30), width = 2, extend = extend.right,
                 style = tlValidate ? line.style_dotted : line.style_solid)
    // order-flow zigzag: leg into this swing low
    if showOF
        if not na(zzPrevP)
            array.push(zzLines, line.new(zzPrevB, zzPrevP, bar_index - pivLen, pl,
                 color = pl > zzPrevP ? color.new(COL_BULL, 35) : color.new(COL_BEAR, 35), width = 1))
            if array.size(zzLines) > 40
                line.delete(array.get(zzLines, 0))
                array.remove(zzLines, 0)
        zzPrevP := pl
        zzPrevB := bar_index - pivLen
    prevPL     := pl
    prevPLBar  := bar_index - pivLen
    lastSup    := pl
    lastSupBar := bar_index - pivLen
    supBroken  := false

// Sweep = wick through the level, close back inside ("sweeping candle").
// A sweep does NOT latch the break — the level stays valid liquidity.
sslSweep = not na(lastSup) and low < lastSup and close > lastSup    // sell-side taken → bullish
bslSweep = not na(lastRes) and high > lastRes and close < lastRes   // buy-side taken → bearish

var int   lastSslBar = na
var int   lastBslBar = na
var float lastSslLvl = na   // FIX-12: the level that sweep took (sweep context)
var float lastBslLvl = na
var bool  lastSslStrong = false   // FIX-21: sweep candle closed in its outer half (quality)
var bool  lastBslStrong = false
if sslSweep and confirmed
    lastSslBar    := bar_index
    lastSslLvl    := lastSup
    lastSslStrong := close >= low + 0.5 * rng
if bslSweep and confirmed
    lastBslBar    := bar_index
    lastBslLvl    := lastRes
    lastBslStrong := close <= high - 0.5 * rng

// EQH/EQL pool maintenance: distinguish Judas sweep (wick through, close back
// inside → high-quality liquidity grab) from a real break (close beyond)
eqhSwept = false
eqlSwept = false
if array.size(eqhLvls) > 0
    for i = array.size(eqhLvls) - 1 to 0
        lvl = array.get(eqhLvls, i)
        if high > lvl and confirmed
            if close < lvl                    // Judas swing above equal highs
                eqhSwept   := true
                lastBslBar := bar_index       // upgrade: pool sweep counts as BSL sweep
                lastBslLvl := lvl
                lastBslStrong := close <= high - 0.5 * rng
            line.delete(array.get(eqhLns, i)) // pool consumed either way
            array.remove(eqhLns, i)
            array.remove(eqhLvls, i)
if array.size(eqlLvls) > 0
    for i = array.size(eqlLvls) - 1 to 0
        lvl = array.get(eqlLvls, i)
        if low < lvl and confirmed
            if close > lvl                    // Judas swing below equal lows
                eqlSwept   := true
                lastSslBar := bar_index
                lastSslLvl := lvl
                lastSslStrong := close >= low + 0.5 * rng
            line.delete(array.get(eqlLns, i))
            array.remove(eqlLns, i)
            array.remove(eqlLvls, i)

recentSsl = not na(lastSslBar) and bar_index - lastSslBar <= sweepLook
recentBsl = not na(lastBslBar) and bar_index - lastBslBar <= sweepLook
// FIX-12: a sweep only backs a signal while the raid is still IN PLAY — price
// has not already travelled away from the level it took. The time gate alone
// credited any sweep to any same-direction signal N bars later, anywhere.
sslInPlay = recentSsl and not na(lastSslLvl) and close - lastSslLvl <= SWEEP_CTX_ATR * atr
bslInPlay = recentBsl and not na(lastBslLvl) and lastBslLvl - close <= SWEEP_CTX_ATR * atr

// FIX-2: first confirmed close beyond the CURRENT level, once per level.
// Also fires when the pivot confirms with price already beyond it (break
// completed during the pivLen confirmation lag — v1 missed those entirely).
brokeUp = not na(lastRes) and not resBroken and close > lastRes
brokeDn = not na(lastSup) and not supBroken and close < lastSup
if brokeUp and confirmed
    resBroken := true
if brokeDn and confirmed
    supBroken := true

// ---- protected-swing structure engine (FIX-13) --------------------------------
// One instance per pivot scale (internal / external). Only a DISPLACED close
// is structural (structDisp). BOS = displaced close through the latest pivot
// in the trend direction (continuation). CHoCH = displaced close through the
// PROTECTED swing = the origin of the leg that produced the last BOS. A run
// through a minor pullback low inside a bull leg is inducement — that pivot
// stays liquidity (sweeps / OTE still use it) but is not structure.
f_structure(bool upBrk, bool dnBrk) =>
    var int   dir    = 0          // 1 bullish, -1 bearish, 0 undefined
    var float protLo = na         // bullish structure's protected low
    var float protHi = na         // bearish structure's protected high
    var float legLo  = na         // lowest low since the last structural event = origin of the current up-leg
    var float legHi  = na
    bool bosU = false
    bool bosD = false
    bool chU  = false
    bool chD  = false
    legLo := na(legLo) ? low : math.min(legLo, low)
    legHi := na(legHi) ? high : math.max(legHi, high)
    if confirmed and body >= structDisp * atr
        if dir == 1
            if upBrk
                bosU   := true
                protLo := legLo
            else if not na(protLo) and close < protLo
                chD    := true
                dir    := -1
                protHi := legHi
                protLo := na
        else if dir == -1
            if dnBrk
                bosD   := true
                protHi := legHi
            else if not na(protHi) and close > protHi
                chU    := true
                dir    := 1
                protLo := legLo
                protHi := na
        else if upBrk
            bosU   := true
            dir    := 1
            protLo := legLo
        else if dnBrk
            bosD   := true
            dir    := -1
            protHi := legHi
        if bosU or bosD or chU or chD          // a new leg starts here
            legLo := low
            legHi := high
    [dir, bosU, bosD, chU, chD, protLo, protHi]

[msDir, bosUp, bosDn, chochUp, chochDn, protLowI, protHighI] = f_structure(brokeUp, brokeDn)

// INTERNAL structure marks — muted chips (minor swings, often inducement)
if showStruct and (bosUp or chochUp)
    f_chip(true, chochUp ? "iCHoCH" : "ibos", color.new(color.gray, 60), color.new(COL_BULL, 0), size.tiny)
if showStruct and (bosDn or chochDn)
    f_chip(false, chochDn ? "iCHoCH" : "ibos", color.new(color.gray, 60), color.new(COL_BEAR, 0), size.tiny)

// ---- EXTERNAL structure engine (ADD-10): major swings, same protected logic ---
eph = ta.pivothigh(high, extPiv, extPiv)
epl = ta.pivotlow(low, extPiv, extPiv)
var float extRes    = na
var float extSup    = na
var int   extResBar = na        // FIX-20: leg direction for the external dealing range
var int   extSupBar = na
var bool  extResBrk = false
var bool  extSupBrk = false
if not na(eph) and confirmed
    extRes    := eph
    extResBar := bar_index - extPiv
    extResBrk := false
if not na(epl) and confirmed
    extSup    := epl
    extSupBar := bar_index - extPiv
    extSupBrk := false
eBrokeUp = not na(extRes) and not extResBrk and close > extRes
eBrokeDn = not na(extSup) and not extSupBrk and close < extSup
if eBrokeUp and confirmed
    extResBrk := true
if eBrokeDn and confirmed
    extSupBrk := true
[extDir, eBosUp, eBosDn, eChochUp, eChochDn, protLowE, protHighE] = f_structure(eBrokeUp, eBrokeDn)

// live trigger memory (FIX-21): a shift / CISD / displacement stays a valid
// trigger for a few bars; structure breaks also feed the FIX-14 breaker rule
var int lastChUpBar   = na
var int lastChDnBar   = na
var int lastBosUpBar  = na
var int lastBosDnBar  = na
var int lastCisdUpBar = na
var int lastCisdDnBar = na
var int lastDispUpBar = na
var int lastDispDnBar = na
if confirmed
    if chochUp or eChochUp
        lastChUpBar := bar_index
    if chochDn or eChochDn
        lastChDnBar := bar_index
    if bosUp or eBosUp
        lastBosUpBar := bar_index
    if bosDn or eBosDn
        lastBosDnBar := bar_index
    if cisdUp
        lastCisdUpBar := bar_index
    if cisdDn
        lastCisdDnBar := bar_index
    if bullBar and body >= dispFactor * atr
        lastDispUpBar := bar_index
    if bearBar and body >= dispFactor * atr
        lastDispDnBar := bar_index
f_recent(int b, int look) => not na(b) and bar_index - b <= look
recentChUp     = f_recent(lastChUpBar, TRIG_LOOK)
recentChDn     = f_recent(lastChDnBar, TRIG_LOOK)
recentBosUp    = f_recent(lastBosUpBar, TRIG_LOOK)
recentBosDn    = f_recent(lastBosDnBar, TRIG_LOOK)
recentCisdUp   = f_recent(lastCisdUpBar, TRIG_LOOK)
recentCisdDn   = f_recent(lastCisdDnBar, TRIG_LOOK)
recentDispUp   = f_recent(lastDispUpBar, 3)
recentDispDn   = f_recent(lastDispDnBar, 3)
recentStructUp = recentChUp or recentBosUp        // FIX-14: breaker needs a structure break
recentStructDn = recentChDn or recentBosDn
// EXTERNAL marks — bright chips: this is the real structure narrative
if showStruct and (eBosUp or eChochUp)
    f_chip(true, eChochUp ? "CHoCH" : "BOS", color.new(COL_BULL, 15), color.white, size.small)
if showStruct and (eBosDn or eChochDn)
    f_chip(false, eChochDn ? "CHoCH" : "BOS", color.new(COL_BEAR, 15), color.white, size.small)

// ═════════════════════════════════════════════════════════════════ STEP 4
// MTF BIAS  (5m / 15m / 1H / 4H / 1D)

f_bias(simple string tf) =>
    request.security(syminfo.tickerid, tf, nz((close > ta.ema(close, biasLen) ? 1 : -1)[1], 0),
         lookahead = barmerge.lookahead_off)

b5   = f_bias("5")

    "5"   => b5
    "15"  => b15
    "60"  => b60

    "D"   => bD
    => 0
htfB = filtBias != 0 ? filtBias : b60      // HTF bias for context scoring (filter TF, else 1H)

inAsia = f_inSes(asiaSes)
inLon  = f_inSes(lonSes)

// ---- HTF CRT + turtle soup (ADD-8) --------------------------------------------
// Chart TF → HTF map per the trading plan: 1m checks 15m, 5m checks 1H,
// 15m checks 4H. All values taken with [1] inside security (last CLOSED
// HTF candle) + lookahead off → non-repainting; fires once per HTF close.
f_autoHtf() =>
    switch timeframe.period
        "1"   => "15"
        "2"   => "30"
        "3"   => "30"
        "5"   => "60"
        "10"  => "120"
        "15"  => "240"
        "30"  => "240"
        "60"  => "D"
        "120" => "D"
        "240" => "D"
        => "D"
htfTf = htfSel == "Auto" ? f_autoHtf() : htfSel
// self-reference guard: HTF logic is meaningless when HTF == chart TF
htfLive = htfTf != timeframe.period

f_crtTbsHtf() =>
    _atr  = ta.atr(atrLen)
    _rng  = high - low
    _body = math.abs(close - open)
    _bull = close > open
    _bear = close < open
    _c1big = _rng[2] >= crtRangeAtr * _atr
    _crtB  = _c1big and low[1] < low[2] and close[1] > low[2] and close[1] < high[2] and low[2] - low[1] >= minWickAtr * _atr and _bull and close > high[1] and _body >= crtDispAtr * _atr
    _crtS  = _c1big and high[1] > high[2] and close[1] < high[2] and close[1] > low[2] and high[1] - high[2] >= minWickAtr * _atr and _bear and close < low[1] and _body >= crtDispAtr * _atr
    _tl    = ta.lowest(low, tbsLen)[1]
    _th    = ta.highest(high, tbsLen)[1]
    _loAge = -ta.lowestbars(low, tbsLen)[1] + 1
    _hiAge = -ta.highestbars(high, tbsLen)[1] + 1
    _tbsB  = low < _tl and close > _tl and _bull and _loAge >= tbsAgeMin and close >= low + 0.5 * _rng
    _tbsS  = high > _th and close < _th and _bear and _hiAge >= tbsAgeMin and close <= high - 0.5 * _rng
    [_crtB[1], _crtS[1], _tbsB[1], _tbsS[1], time[1]]

[hCrtB, hCrtS, hTbsB, hTbsS, hT] = request.security(syminfo.tickerid, htfTf, f_crtTbsHtf(),
     lookahead = barmerge.lookahead_off)

    var float sBot = na
    var int   sBar = na

        bTop := low
        bBot := high[2]
        bBar := bar_index
        bIn  := 0
    if high < low[2]                       // bearish FVG completed on HTF
        sBot := high
        sTop := low[2]
        sBar := bar_index
        sIn  := 0
    if not na(bTop) and bar_index > bBar and low <= bTop
        bIn += 1
    if not na(sTop) and bar_index > sBar and high >= sBot
        sIn += 1
    // FIX-15: dead when disrespected (inversion case), too old, or consumed
    if not na(bBot) and (close < bBot or bar_index - bBar > hPoiMaxAge or bIn > hPoiMaxTouch)
        bTop := na
        bBot := na
    if not na(sTop) and (close > sTop or bar_index - sBar > hPoiMaxAge or sIn > hPoiMaxTouch)
        sTop := na
        sBot := na
    [bTop[1], bBot[1], sTop[1], sBot[1], bIn[1], sIn[1]]

shrankPrev = false
if array.size(wicks) >= 3
    wA = array.get(wicks, array.size(wicks) - 3)
    wB = array.get(wicks, array.size(wicks) - 2)

    shrinking := wA > wB and wB > wC and wA >= minWickAtr * atr
if array.size(wicks) >= 4
    pA = array.get(wicks, array.size(wicks) - 4)
    pB = array.get(wicks, array.size(wicks) - 3)
    pC = array.get(wicks, array.size(wicks) - 2)
    grewPrev   := pC > pB and pB > pA and pB >= minWickAtr * atr
    shrankPrev := pA > pB and pB > pC and pA >= minWickAtr * atr

if wState == 1 and confirmed
    if showS1 and (growing or grewPrev) and bearBar and close < wLevel and body >= BRK_BODY_ATR * atr
        sig1Short := true                 // ① sellers stacked, closed back below level
        sigSLShort := wExtreme + SL_PAD_ATR * atr    // SL above the highest trap wick
        wState := 0
    else if showS2 and (shrinking or shrankPrev) and bullConfirm
        sig2Long := true                  // ② early shorts trapped above their pins
        wState := 0
    else if bar_index - wStart > maxAfterBrk or close > wLevel + RUNAWAY_ATR * atr
        wState := 0
else if wState == -1 and confirmed
    if showS1 and (growing or grewPrev) and bullBar and close > wLevel and body >= BRK_BODY_ATR * atr
        sig1Long := true                  // ① buyers stacked, closed back above level
        sigSLLong := wExtreme - SL_PAD_ATR * atr    // SL below the lowest trap wick
        wState := 0
    else if showS2 and (shrinking or shrankPrev) and bearConfirm
        sig2Short := true                 // ② early longs trapped below their pins
        wState := 0
    else if bar_index - wStart > maxAfterBrk or close < wLevel - RUNAWAY_ATR * atr
        wState := 0

// SETUP ③ — INSIDE BAR BREAKOUT (50% mother-candle filter)

            ibHigh := high
            ibLow  := low
            ibStage := 2
        else
            ibStage := 0
    else if ibStage == 2

        // HTF bias) and a displacement close — a weak tick through the IB high
        // is a failed setup, not a late entry

            ibStage := 0
        else if close < ibLow or bar_index - ibStart > IB_WAIT_BARS

    else if ibStage == -2

                sig3Short := showS3
            ibStage := 0
        else if close > ibHigh or bar_index - ibStart > IB_WAIT_BARS

        mLow  := low
